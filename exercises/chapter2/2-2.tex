\subsection{Centralizers and Normalizers, Stabilizers and Kernels}

\begin{exercise}{2.2.1}
    Prove that $C_G(A)=\set{g\in G\mid g\inv ag=a\text{ for all }a\in A}.$
\end{exercise}

\begin{sol}
    $g \in C_G(A) \iff ag = ga$ for all $a \in A \iff g\inv ag = a$ for all $a \in A$.
\end{sol}

\begin{exercise}{2.2.2}
    Prove that $C_G(Z(G))=G$ and deduce that $N_G(Z(G))=G.$
\end{exercise}

\begin{sol}
	Let $g \in G$ and $z \in Z(G)$.
	Then $gz = zg$, so $g \in C_G(Z(G))$.
    Therefore, $G \subseteq C_G(Z(G))$, so $C_G(Z(G)) = G$.
	Since $C_G(Z(G)) \leq N_G(Z(G)) \leq G$, $N_G(Z(G)) = G$.
\end{sol}

\begin{exercise}{2.2.3}
    Prove that if $A$ and $B$ are subsets of $G$ with $A \subseteq B$ then $C_G(B)$ is a subgroup of $C_G(A)$.
\end{exercise}

\begin{sol}
    Pick $g \in C_G(B)$, so $gbg\inv = b$ for all $b \in B$.
	In particular, $gag\inv = a$ for all $a \in A$ because $A \subseteq B$, so $g \in C_G(A)$, and thus $C_G(B) \subseteq C_G(A)$.
	Since $C_G(B)$ and $C_G(A)$ are subgroups of $G$, $C_G(B) \leq C_G(A)$.
\end{sol}

\begin{exercise}!{2.2.4}
    For each of $S_3$, $D_8$, and $Q_8$ compute the centralizers of each element and find the center of each group.
	Does Lagrange's Theorem (\exref{1.7.19}) simplify your work?
\end{exercise}

\begin{sol}
    Observe that $Z(G) = \bigcap_{g \in G} C_G(g)$ and $C_G(1) = G$.
	For $S_3$, let $C = C_{S_3}(\cyc{1 2})$.
    Since $\cyc{1 2}\cyc{1 3} \neq \cyc{1 3}\cyc{1 2}$, we have $C \neq S_3$.
    Because $\set{1, \cyc{1 2}} \leq C$ and $C \leq S_3$, Lagrange's Theorem gives $\abs{C} \mid 6$ and $2 \mid \abs{C}$, so $\abs{C} = 2$, and therefore $C = \set{1, \cyc{1 2}}$.
	By symmetry, the same argument holds for $\cyc{1 3}$ and $\cyc{2 3}$, and thus
    \[
        C_{S_3}(\cyc{1 2}) = \set{1, \cyc{1 2}}, \quad 
    	C_{S_3}(\cyc{1 3}) = \set{1, \cyc{1 3}}, \quad 
        C_{S_3}(\cyc{2 3}) = \set{1, \cyc{2 3}}.
    \]
    A similar argument for $\cyc{1 2 3}$, along with $\cyc{1 2 3}\cyc{1 2} \neq \cyc{1 2}\cyc{1 2 3}$, shows that $C_{S_3}(\cyc{1 2 3}) = \set{1, \cyc{1 2 3}, \cyc{1 3 2}}$.
    Since $C_G(x) = C_G(x\inv)$, we have $C_{S_3}(\cyc{1 3 2}) = C_{S_3}(\cyc{1 2 3})$.
    Finally, $Z(S_3) = \set{1}$.
    
    Note that $r^2$ commutes with $r$ and $s$, so $r^2 \in Z(D_8)$, and the centralizers of $D_8$ are:
    \begin{align*}
        C_{D_8}(1) &= D_8 & C_{D_8}(s) &= \set{1, r^2, s, sr^2} \\
        C_{D_8}(r) &= \set{1, r, r^2, r^3} & C_{D_8}(sr) &= \set{1, r^2, sr, sr^3} \\
        C_{D_8}(r^2) &= D_8 & C_{D_8}(sr^2) &= \set{1, r^2, s, sr^2} \\
        C_{D_8}(r^3) &= \set{1, r, r^2, r^3} & C_{D_8}(sr^3) &= \set{1, r^2, sr, sr^3},
    \end{align*}
    and $Z(D_8) = \set{1, r^2}$.
	Now, $-1$ commutes with all elements of $Q_8$, so $-1 \in Z(Q_8)$, and the centralizers of $Q_8$ are:
    \begin{align*}
        C_{Q_8}(1) & = Q_8 & C_{Q_8}(-1) & = Q_8 \\
        C_{Q_8}(i) & = \set{1, -1, i, -i} & C_{Q_8}(-i) & = \set{1, -1, i, -i} \\
        C_{Q_8}(j) & = \set{1, -1, j, -j} & C_{Q_8}(-j) & = \set{1, -1, j, -j} \\
        C_{Q_8}(k) & = \set{1, -1, k, -k} & C_{Q_8}(-k) & = \set{1, -1, k, -k}
    \end{align*}
    and $Z(Q_8) = \set{1, -1}$.

    Let $G$ be any one of $S_3$, $D_8$, or $Q_8$.
    By \exref{1.1.35}, the cyclic subgroup generated by any $g \in G$ has order $\abs{g}$.
    Since the cyclic subgroup generated by $g$ and $Z(G)$ are both contained in $C_G(g)$, Lagrange's Theorem gives $\lcm(\abs{g}, \abs{Z(G)}) \mid \abs{C_G(g)}$.
    Since $\abs{C_G(g)} \mid \abs{G}$ as well, finding an element that does not commute with noncentral $g$ determines the exact order of $C_G(g)$.
\end{sol}

\begin{exercise}{2.2.5}
    In each of parts (a) to (c) show that for the specified group $G$ and subgroup $A$ of $G$, $C_G(A)=A$ and $N_G(A)=G.$
    \begin{subexercises}
        \item $G=S_3$ and $A=\set{1,\cyc{1 2 3},\cyc{1 3 2}}.$
        \item $G=D_8$ and $A=\set{1, s, r^2, sr^2}.$
        \item $G=D_{10}$ and $A=\set{1,r,r^2,r^3,r^4}.$
    \end{subexercises}
\end{exercise}

\begin{sol}
    Each part will verify $C_G(A) = A$.
    To show $N_G(A) = G$, we use the fact that $C_G(A) \leq N_G(A)$ and that $\abs{G}/\abs{C_G(A)} = 2$, so $\abs{N_G(A)}$ must be either $\abs{C_G(A)}$ or $\abs{G}$ by Lagrange's Theorem.
    We then give some $x \in G - A$ such that $xAx\inv = A$, so $x \in N_G(A)$ and thus $N_G(A) = G$.
    \begin{subexercises}
        \item By \exref{2.2.4}, $C_{S_3}(A) = C_{S_3}(\cyc{1 2 3}) \cap C_{S_3}(\cyc{1 3 2}) = A$.
        Use $x = \cyc{1 2}$.
        \item By \exref{2.2.4}, $C_{D_8}(A) = C_{D_8}(s) \cap C_{D_8}(r^2) \cap C_{D_8}(sr^2) = A$.
        Use $x = r$.
        \item Observe that $A$ is the cyclic subgroup generated by $r$.
        By Lagrange's Theorem, $5 \mid \abs{C_{D_{10}}(A)}$ and $\abs{C_{D_{10}}(A)} \mid 10$.
        Because $sr \neq rs$, we have $s \notin C_{D_{10}}(A)$, so $C_{D_{10}}(A) = A$.
        Use $x = s$. \qedhere
    \end{subexercises}
\end{sol}

\begin{exercise}{2.2.6}
    Let $H$ be a subgroup of the group $G$.
    \begin{subexercises}
        \item Show that $H\leq N_G(H)$.
        Give an example to show that this is not necessarily true if $H$ is not a subgroup.
        \item Show that $H\leq C_G(H)$ if and only if $H$ is abelian.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Let $h \in H$.
    Closure gives $hHh\inv \subseteq H$.
    The same argument applied to $h\inv$ gives $h\inv Hh \subseteq H$, so $hHh\inv = H$ and $h \in N_G(H)$.

    As a counterexample, let $G = D_6$ and $H = \set{1, r, s}$.
	Then
    \[
    	rHr\inv = \set{1, r, sr} \neq H
    \]
    so that $r$ does not normalize $H$, and thus $H \nsubseteq N_G(H)$.
    \item $H \leq C_G(H) \iff hk = kh$ for all $h, k \in H \iff H$ is abelian.
\end{solenum}

\begin{exercise}{2.2.8}
    Let $G = S_n$, fix an $i \in \set{1, 2, \dots, n}$ and let $G_i=\set{\sigma\in G\mid\sigma(i)=i}$ (the stabilizer of $i$ in $G$).
	Use group actions to prove that $G_i$ is a subgroup of $G$.
	Find $\abs{G_i}$.
\end{exercise}

\begin{sol}
    Since $S_n$ acts on $\set{1, 2, \ldots, n}$ by permutation, the stabilizer of $i$ under this action is a subgroup of $S_n$, so $G_i \leq G$.
    For any $\sigma \in G_i$, we have $\sigma(i) = i$ and $\sigma$ injective, so $\sigma$ permutes $\set{1, 2, \ldots, n} - \set{i}$.
    Conversely, any such permutation of $\set{1, 2, \ldots, n} - \set{i}$ extends to a permutation in $G_i$ by fixing $i$.
    There are $(n - 1)!$ ways to permute these $n - 1$ elements, so $\abs{G_i} = (n - 1)!$.
\end{sol}

\begin{exercise}{2.2.9}
    For any subgroup $H$ of $G$ and any nonempty subset $A$ of $G$ define $N_H(A)$ to be the set $\set{h \in H \mid hAh\inv = A}$.
	Show that $N_H(A) = N_G(A) \cap H$ and deduce that $N_H(A)$ is a subgroup of $H$ (note that $A$ need not be a subset of $H$).
\end{exercise}

\begin{sol}
    We have $N_H(A) = \set{h \in H \mid hAh\inv = A} = \set{g \in G \mid gAg\inv = A} \cap H = N_G(A) \cap H$.
    By \exref{2.1.10}, $N_G(A) \cap H \leq G$, and since $N_G(A) \cap H \subseteq H$, we have $N_G(A) \cap H \leq H$.
\end{sol}

\begin{exercise}{2.2.10}
    Let $H$ be a subgroup of order $2$ in $G$.
	Show that $N_G(H) = C_G(H)$.
	Deduce that if $N_G(H) = G$ then $H \leq Z(G)$.
\end{exercise}

\begin{sol}
    By assumption, $H = \set{1, h}$ for some $h \in G$ with $h \neq 1$.
	By the text, $C_G(H) \leq N_G(H)$.
    Pick $g \in N_G(H)$, so $gHg\inv = H$.
    Then
    \[
        gHg\inv = \set{g1g\inv, ghg\inv} = \set{1, ghg\inv} = H = \set{1, h}.
    \]
    Observe $ghg\inv \neq 1$, for otherwise $gh = g$, so $h = 1$, contradicting that $h \neq 1$.
    Then $ghg\inv = h$, so $g \in C_G(H)$, and hence $N_G(H) \subseteq C_G(H)$, whence $C_G(H) = N_G(H)$.
    If $N_G(H) = G$, then $C_G(H) = G$, so every element of $H$ commutes with every $g \in G$, and therefore $H \leq Z(G)$.
\end{sol}

\begin{exercise}{2.2.11}
    Prove that $Z(G) \leq N_G(A)$ for any subset $A$ of $G$.
\end{exercise}

\begin{sol}
    Note that $Z(G) \leq C_G(A)$ and the text gives $C_G(A) \leq N_G(A)$, so $Z(G) \leq N_G(A)$.
\end{sol}

\begin{exercise}{2.2.12}
    Let $R$ be the set of all polynomials with integer coefficients in the independent variables $x_1, x_2, x_3, x_4$ i.e., the members of $R$ are finite sums of elements of the form $ax_1^{r_1}x_2^{r_2}x_3^{r_3}x_4^{r_4}$, where $a$ is any integer and $r_1, \ldots, r_4$ are nonnegative integers.
	For example,
    \begin{equation}
        \label{ex2.2.12eq}
        \tag{$\ast$}
        12x_1^5x_2^7x_4 - 18x_2^3x_3 + 11x_1^6x_2x_3^3x_4^{23}
    \end{equation}
    is a typical element of $R$.
	Each $\sigma \in S_4$ gives a permutation of $\set{x_1, \ldots, x_4}$ by defining $\sigma \cdot x_i = x_{\sigma(i)}$.
	This may be extended to a map from $R$ to $R$ by defining
    \[
    	\sigma \cdot p(x_1, x_2, x_3, x_4) = p(x_{\sigma(1)}, x_{\sigma(2)}, x_{\sigma(3)}, x_{\sigma(4)})
    \]
    for all $p(x_1, x_2, x_3, x_4) \in R$ (i.e., $\sigma$ simply permutes the indices of the variables).
    For example, if $\sigma = \cyc{1 2}\cyc{3 4}$ and $p(x_1, \ldots, x_4)$ is the polynomial in \eqref{ex2.2.12eq} above, then
    \begin{align*}
        \sigma \cdot p(x_1, x_2, x_3, x_4) &= 12x_2^5x_1^7x_3 - 18x_1^3x_4 + 11x_2^6x_1x_4^3x_3^{23} \\
        &= 12x_1^7x_2^5x_3 - 18x_1^3x_4 + 11x_1x_2^6x_3^{23}x_4^3.
    \end{align*}
    \begin{subexercises}
        \item Let $p = p(x_1, \ldots, x_4)$ be the polynomial in \eqref{ex2.2.12eq} above, let $\sigma = \cyc{1 2 3 4}$ and let $\tau = \cyc{1 2 3}$.
	    Compute $\sigma \cdot p, \tau \cdot (\sigma \cdot p), (\tau \circ \sigma) \cdot p$, and $(\sigma \circ \tau) \cdot p$.
        \item Prove that these definitions give a (left) group action of $S_4$ on $R$.
        \item Exhibit all permutations in $S_4$ that stabilize $x_4$ and prove that they form a subgroup isomorphic to $S_3$.
        \item Exhibit all permutations in $S_4$ that stabilize the element $x_1 + x_2$ and prove that they form an abelian subgroup of order 4.
        \item Exhibit all permutations in $S_4$ that stabilize the element $x_1x_2 + x_3x_4$ and prove that they form a subgroup isomorphic to the dihedral group of order 8.
        \item Show that the permutations in $S_4$ that stabilize the element $(x_1 + x_2)(x_3 + x_4)$ are exactly the same as those found in part (e).
        \note{The two polynomials appearing in parts (e) and (f) and the subgroup that stabilizes them will play an important role in the study of roots of quartic equations in Section 14.6.}
    \end{subexercises}
\end{exercise}

\begin{sol}
    For parts (c) -- (f), let $q$ be the element to be stabilized.
    The text gives that the stabilizer of any $q \in R$ is a subgroup of $S_4$, so it remains to be shown that the specific stabilizers have the claimed structure.
    \begin{subexercises}
        \item Note that $\tau \circ \sigma = \cyc{1 3 4 2}$ and $\sigma \circ \tau = \cyc{1 3 2 4}$.
        Then
        \begin{align*}
            \sigma \cdot p & = 12x_1x_2^5x_3^7 - 18x_3^3x_4 + 11x_1^{23}x_2^6x_3x_4^3 \\
            \tau \cdot (\sigma \cdot p) & = 12x_1^7x_2x_3^5 - 18x_1^3x_4 + 11x_1x_2^{23}x_3^6x_4^3 \\
            (\tau \circ \sigma) \cdot p & = 12x_1^7x_2x_3^5 - 18x_1^3x_4 + 11x_1x_2^{23}x_3^6x_4^3  \\
            (\sigma \circ \tau) \cdot p & = 12x_1x_3^5x_4^7 - 18x_2x_4^3 + 11x_1^{23}x_2^3x_3^6x_4
        \end{align*}
        \item Note that $1 \cdot r_1 = r_1$ for any $r_1 \in R$.
        Let $\alpha, \beta \in S_4$, and put $r_2 = \beta \cdot r_1(x_1, x_2, x_3, x_4)$.
        Then $\alpha \cdot r_2$ replaces each $x_j$ in $r_2$ by $x_{\alpha(j)}$, so $x_{\beta(j)}$ becomes $x_{\alpha(\beta(j))}$.
        Therefore,
        \begin{align*}
            \alpha \cdot r_2 & = \alpha \cdot (\beta \cdot r_1(x_1, x_2, x_3, x_4)) \\
            & = \alpha \cdot r_1(x_{\beta(1)}, x_{\beta(2)}, x_{\beta(3)}, x_{\beta(4)}) \\
            & = r_1(x_{\alpha(\beta(1))}, x_{\alpha(\beta(2))}, x_{\alpha(\beta(3))}, x_{\alpha(\beta(4))}) \\
            & = (\alpha \circ \beta) \cdot r_1(x_1, x_2, x_3, x_4)
        \end{align*}
        \item We have $\alpha \in G_q$ if and only if $\alpha(4) = 4$, so $G_q$ consists of the permutations of $\set{1, 2, 3}$ extended by fixing 4:
        \[
            G_q = \set{1, \cyc{1 2}, \cyc{1 3}, \cyc{2 3}, \cyc{1 2 3}, \cyc{1 3 2}}.
        \]
        Restricting $\alpha$ to $\set{1, 2, 3}$ gives a bijection $G_q \to S_3$ defined by $\alpha \mapsto \alpha|_{\set{1, 2, 3}}$.
        Moreover, for $\alpha, \beta \in G_q$, we have $(\alpha \circ \beta)|_{\set{1, 2, 3}} = \alpha|_{\set{1, 2, 3}} \circ \beta|_{\set{1, 2, 3}}$, so the bijection is a homomorphism, and thus $G_q \cong S_3$.
        \item We have $\alpha \in G_q$ if and only if $\alpha(\set{1, 2}) = \set{1, 2}$.
        These are exactly the permutations
        \[
            G_q = \set{1, \cyc{1 2}, \cyc{3 4}, \cyc{1 2}\cyc{3 4}}.
        \]
        All nonidentity elements of $G_q$ have order 2, so it is abelian by \exref{1.1.25}.
        \item We have $\alpha \in G_q$ if and only if $\alpha(\set{\set{1, 2}, \set{3, 4}}) = \set{\set{1, 2}, \set{3, 4}}$, so the permutations that satisfy this condition are exactly
        \[
            G_q = \set{1, \cyc{1 2}, \cyc{3 4}, \cyc{1 2}\cyc{3 4}, \cyc{1 3}\cyc{2 4}, \cyc{1 4}\cyc{2 3}, \cyc{1 3 2 4}, \cyc{1 4 2 3}}.
        \]
        Put $\pi = \cyc{1 2}$ and $\rho = \cyc{1 3 2 4}$.
        The eight elements $\rho^k$ and $\pi \rho^k$ for $0 \leq k \leq 3$ are all distinct and account for all elements of $G_q$.
        By inspection, $\rho^4 = \pi^2 = 1$, and
        \[
            \rho \pi = \cyc{1 3 2 4}\cyc{1 2} = \cyc{1 4}\cyc{2 3} = \cyc{1 2}\cyc{1 4 2 3} = \pi \rho\inv.
        \]
        Since $\rho$ and $\pi$ satisfy the same relations as those in $D_8$, there exists a surjective homomorphism $\phi \colon D_8 \to G_q$ defined by $\phi(r) = \rho$ and $\phi(s) = \pi$.
        Since $\abs{D_8} = \abs{G_q} = 8$, $\phi$ is an isomorphism, so $G_q \cong D_8$.
        \item Rewrite $q$ as 
        \[
            q = \sum_{i < j} x_ix_j - (x_1x_2 + x_3x_4) = q_0 - (x_1x_2 + x_3x_4).
        \]
        For any $\alpha \in S_4$, we have $\alpha(\set{x_ix_j \mid i < j}) = \set{x_{\alpha(i)}x_{\alpha(j)} \mid i < j} = \set{x_ix_j \mid i < j}$, so $\alpha \cdot q_0 = q_0$.
        Moreover, for any $f, g \in R$, we have $\alpha \cdot(f + g) = \alpha \cdot f + \alpha \cdot g$, so $\alpha \cdot q = q_0 - \alpha \cdot (x_1x_2 + x_3x_4)$.
        Therefore, $\alpha \cdot q = q$ if and only if $\alpha$ fixes $x_1x_2 + x_3x_4$, which is exactly the stabilizer obtained in part (e). \qedhere
    \end{subexercises}
\end{sol}

\begin{exercise}{2.2.14}
    Let $H(F)$ be the Heisenberg group over the field $F$ introduced in \exref{1.4.11}.
	Determine which matrices lie in the center of $H(F)$ and prove that $Z(H(F))$ is isomorphic to the additive group $F$.
\end{exercise}

\begin{sol}
    Let $X = (a, b, c) \in Z(H(F))$ and $Y = (d, e, f) \in H(F)$.
	Then
    \[
        XY = (a + d, b + e + af, c + f), \quad
        YX = (d + a, e + b + dc, f + c).
    \]
    Since $XY = YX$, the middle entries must be equal, so $af = dc$ for all $d, f \in F$.
    In particular, $d = 0$ and $f = 1$ gives $a = 0$, and $d = 1$ and $f = 0$ gives $c = 0$.
	Conversely, any $(0, z, 0) \in H(F)$ commutes with all elements of $H(F)$, so it lies in the center.
    Therefore,
    \[
    	Z(H(F)) = \set{(0, z, 0) \mid z \in F}.
    \]
    The map $z \mapsto (0, z, 0)$ has a two-sided inverse given by $(0, z, 0) \mapsto z$, so it is a bijection.
    Moreover, for any $x, y \in F$, we have
    \[
        (0, x, 0)(0, y, 0) = (0, x + y, 0),
    \]
    which shows that the map is a homomorphism.
    Therefore, it is an isomorphism, and $Z(H(F)) \cong F$.
\end{sol}
