\subsection{Group Actions}

\begin{exercise}{1.7.1}
    Let $F$ be a field.
	Show that the multiplicative group of nonzero elements of $F$ (denoted by $F\unt$) acts on the set $F$ by $g\cdot a = g a$, where $g\in F\unt$, $a\in F$ and $ga$ is the usual product in $F$ of the two field elements (state clearly which axioms in the definition of a field are used). 
\end{exercise}

\begin{sol}
    Let $g, h \in F\unt$ and $a \in F$.
	The action is well defined by closure of multiplication in $F$.
    Then
    \[
    	g \cdot (h \cdot a) = g \cdot (ha) = g(ha) = (gh)a = (gh) \cdot a,
    \]
    where the associativity of multiplication in $F$ was used to conclude that $g(ha) = (gh)a$.
	Moreover, $1 \cdot a = 1a = a$ by the multiplicative identity axiom in $F$.
	Therefore, the group action axioms are satisfied and $F\unt$ acts on $F$.
\end{sol}

\begin{exercise}{1.7.2}
    Show that the additive group $\zbb$ acts on itself by $z\cdot a = z + a$ for all $z,a\in\zbb$.
\end{exercise}

\begin{sol}
    Let $z, w \in \zbb$ and $a \in \zbb$.
	Then
    \[
    	z \cdot (w \cdot a) = z \cdot (w + a) = z + (w + a) = (z + w) + a = (z + w) \cdot a.
    \]
    Moreover, $0 \cdot a = 0 + a = a$, so $\zbb$ acts on itself.
\end{sol}

\begin{exercise}{1.7.3}
    Show that the additive group $\rbb$ acts on the $x,y$ plane $\rbb\times\rbb$ by $r\cdot (x,y) = (x + r, y)$.
\end{exercise}

\begin{sol}
    Let $r, s \in \rbb$ and $(x, y) \in \rbb^2$.
	Then
    \[
    	r \cdot (s \cdot (x, y)) = r \cdot (x + s, y) = (x + s + r, y) = (x + (r + s), y) = (r + s) \cdot (x, y).
    \]
    Moreover, $0 \cdot (x, y) = (x + 0, y) = (x, y)$, and thus $\rbb$ acts on $\rbb^2$.
\end{sol}

\begin{exercise}![Kernel and Stabilizer of a Group Action]{1.7.4}
    Let $G$ be a group acting on a set $A$ and fix some $a\in A$.
	Show that the following sets are subgroups of $G$:
    \begin{subexercises}
        \item the kernel of the action,
        \item $\set{g\in G \mid g a = a}$---this subgroup is called the \term{stabilizer} of $a$ in $G$.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Let $K$ denote the kernel of the action and let $b \in A$.
    Since $1 \cdot b = b$, we have $1 \in K$.
    Then for any $g, h \in K$, we have
    \[
        gh \cdot b = g \cdot (h \cdot b) = g \cdot b = b, \quad g\inv \cdot b = g\inv \cdot (g \cdot b) = (g\inv g) \cdot b = 1 \cdot b = b,
    \]
    and therefore $gh, g\inv \in K$.
    Hence, $K$ is a subgroup of $G$.
    \item Let $G_a$ denote the stabilizer of $a$ in $G$.
	The same computations as in part (a) with $b = a$ show that $G_a$ is a subgroup of $G$.
\end{solenum}

\begin{exercise}{1.7.5}
    Prove that the kernel of an action of the group $G$ on the set $A$ is the same as the kernel of the corresponding permutation representation $G\to S_A$ (cf. \exref{1.6.14}).
\end{exercise}

\begin{sol}
    Let $K$ denote the kernel of the action and let $\phi \colon G \to S_A$ be the permutation representation corresponding to the action, so $\phi(g)(a) = g \cdot a$ for all $g \in G$ and $a \in A$.
    Then
    \[
        g \in K \iff g \cdot a = a \text{ for all } a \in A \iff \phi(g) = 1 \iff g \in \ker \phi. \qedhere
    \]
\end{sol}

\begin{exercise}{1.7.6}
    Prove that a group $G$ acts faithfully on a set $A$ if and only if the kernel of the action is the set consisting only of the identity.
\end{exercise}

\begin{sol}
    By definition, the action is faithful if and only if its permutation representation $\phi \colon G \to S_A$ is injective.
    By \exref{1.6.14}, this holds if and only if $\ker \phi = 1$, which by \exref{1.7.5} is equivalent to the kernel of the action being the set consisting only of the identity.
\end{sol}

\begin{exercise}{1.7.7}
    Prove that in Example 2 in this section the action is faithful.
\end{exercise}

\begin{sol}
    Let $V$ be a vector space over a field $F$.
    If $V = \set{0}$ and $\abs{F} > 2$, then $\abs{F\unt} > 1$, so there exists $r \in F\unt$ with $r \neq 1$ such that $r \cdot 0 = 0$, and thus the action is not faithful.
    If $\abs{F} = 2$, then $F\unt = \set{1}$, so the action is faithful.
    Otherwise, $V \neq \set{0}$, so we may pick $v \in V$ with $v \neq 0$.
    Suppose $r \cdot v = v$ for some $r \in F\unt$.
    Then $rv = v$, so $(r - 1)v = 0$.
    Since $v \neq 0$ and $F$ is a field, $r = 1$, so the action is faithful by \exref{1.7.6}.
\end{sol}

\begin{exercise}{1.7.8}
    Let $A$ be a nonempty set and let $k$ be a positive integer with $k \leq \abs{A}$.
	The symmetric group $S_A$ acts on the set $B$ consisting of all subsets of $A$ of cardinality $k$ by
    \[
    	\sigma(\set{a_1, \dots, a_k}) = \set{\sigma(a_1), \dots, \sigma(a_k)}.
    \]
    \begin{subexercises}
        \item Prove that this is a group action.
        \item Describe explicitly how the elements $\cyc{1 2}$ and $\cyc{1 2 3}$ act on the six $2$-element subsets of $\set{1,2,3,4}$.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Let $\sigma, \tau \in S_A$ and let $S = \set{x_1, \ldots, x_k} \in B$.
    Since $\sigma$ is injective, $\abs{\sigma(S)} = \abs{S} = k$, so $\sigma \cdot S \in B$.
	Then
    \begin{align*}
        \sigma \cdot (\tau \cdot S) & = \sigma \cdot \set{\tau(x_1), \ldots, \tau(x_k)} \\
        & = \set{\sigma(\tau(x_1)), \ldots, \sigma(\tau(x_k))} \\
        & = (\sigma \circ \tau) \cdot S.
    \end{align*}
    Moreover, $1 \cdot S = \set{1(x_1), \ldots, 1(x_k)} = S$, so $S_A$ acts on $B$.
    \item
    \[
        \begin{array}{*{2}{c|}c}
            S & \cyc{1 2} \cdot S & \cyc{1 2 3} \cdot S \\
            \hline
            \set{1, 2} & \set{1, 2} & \set{2, 3} \\
            \set{1, 3} & \set{2, 3} & \set{1, 2} \\
            \set{1, 4} & \set{2, 4} & \set{2, 4} \\
            \set{2, 3} & \set{1, 3} & \set{1, 3} \\
            \set{2, 4} & \set{1, 4} & \set{3, 4} \\
            \set{3, 4} & \set{3, 4} & \set{1, 4}
        \end{array} \qedhere
    \]
\end{solenum}

\begin{exercise}{1.7.10}
    With reference to the preceding two exercises determine:
    \begin{subexercises}
        \item for which values of $k$ the action of $S_n$ on $k$-element subsets is faithful, and
        \item for which values of $k$ the action of $S_n$ on ordered $k$-tuples is faithful.
    \end{subexercises}
\end{exercise}

\begin{sol}
    Let $A = \set{1, \ldots, n}$.
    If $n = 1$, then $S_n$ is trivial, so the action is faithful for all $k$.
    Assume $n > 1$ and let $\sigma \in S_n$ be nonidentity.
    Then there exists $a \in A$ such that $\sigma(a) \neq a$.
    \begin{subexercises}
        \item If $k < n$, then $\abs{A - \set{\sigma(a)}} = n - 1 \geq k$, so there exists $S \subseteq A$ with $\abs{S} = k$ and $\sigma(a) \notin S$ but $a \in S$.
        Then $\sigma(S) \neq S$, so the action is faithful for $k < n$.
        If $k = n$, then the only $k$-element subset of $A$ is $A$ itself, which is fixed by all permutations, so the action is not faithful for $k = n$.
        
        \item Consider the ordered $k$-tuple $T = (a, x_2, \ldots, x_k)$, where $x_2, \ldots, x_k \in A \setminus \set{a}$ are distinct and exist since $k \leq n$.
        Then $\sigma(T) = (\sigma(a), \sigma(x_2), \ldots, \sigma(x_k)) \neq T$ because $\sigma(a) \neq a$.
        Therefore, no nonidentity permutation fixes every ordered $k$-tuple of elements of $A$, and the action is faithful for all $1 \leq k \leq n$. \qedhere
    \end{subexercises}
\end{sol}

\begin{exercise}{1.7.11}
    Write out the cycle decomposition of the eight permutations in $S_4$ corresponding to the elements of $D_8$ given by the action of $D_8$ on the vertices of a square (where the vertices of the square are labeled as in Section 2).
\end{exercise}

\begin{sol}
    Let $\phi \colon D_8 \to S_4$ be the permutation representation of the action of $D_8$ on the vertices of a square $\set{1, 2, 3, 4}$, where the vertices are labeled as in Section 1.2.
	Then the cycle decompositions are as follows:
    \begin{align*}
        \phi(1) & = 1 & \phi(s) & = \cyc{2 4} \\
        \phi(r) & = \cyc{1 2 3 4} & \phi(sr) & = \cyc{1 4}\cyc{2 3} \\
        \phi(r^2) & = \cyc{1 3}\cyc{2 4} & \phi(sr^2) & = \cyc{1 3} \\
        \phi(r^3) & = \cyc{1 4 3 2} & \phi(sr^3) & = \cyc{1 2}\cyc{3 4}. \qedhere
    \end{align*}
\end{sol}

\begin{exercise}{1.7.12}
    Assume $n$ is an even positive integer and show that $D_{2n}$ acts on the set consisting of pairs of opposite vertices of a regular $n$-gon.
	Find the kernel of this action (label vertices as usual).
\end{exercise}

\begin{sol}
    For convenience, label the vertices with $0, 1, \ldots, n - 1$ clockwise and let $n \geq 4$ be even.
    Let $r$ denote the rotation from vertex $0$ to vertex $1$ and let $s$ denote the reflection across the line from the center to vertex $0$, so $r(j) = j + 1$ and $s(j) = -j$ for all $j$ with vertex labels read mod $n$.
    Let $P_k$ denote the $k$th pair of opposite vertices
    \[
        P_k = \set{k, k + n/2} \quad \text{for } k = 0, \ldots, n/2 - 1,
    \]
    where the subscripts are read mod $n/2$.
    Then for any $j$, $r(j + n/2) = j + n/2 + 1 = r(j) + n/2$ and $s(j + n/2) = -j - n/2 \equiv s(j) + n/2 \pmod{n}$, so $r$ and $s$ indeed map opposite pairs to opposite pairs.
    Thus, for any $g \in D_{2n}$, define $g \cdot \set{a, b} = \set{g(a), g(b)}$, which is a group action by \exref{1.7.8}(a).
    
    For all $i$ and $k$, we have $r^k \cdot P_i = P_{i + k}$ and $sr^k \cdot P_i = P_{-(i + k)}$ with subscripts read mod $n/2$.
    Thus, $r^k \cdot P_i = P_i$ if and only if $k \equiv 0 \pmod{n/2}$ and $sr^k \cdot P_i = P_i$ if and only if $2i + k \equiv 0 \pmod{n/2}$ for all $i$.
    Taking $i = 0$ gives $k \equiv 0 \pmod{n/2}$, while $i = 1$ gives $n/2 \mid 2$, so $n = 4$.
    Conversely, $n = 4$ and $k$ even satisfy $2i + k \equiv 0 \pmod{2}$ for all $i$, so the kernel of the action is $\set{1, r^2, s, sr^2}$, and for $n \geq 6$, the kernel is $\set{1, r^{n/2}}$.
\end{sol}

\begin{exercise}{1.7.13}
    Find the kernel of the left regular action.
\end{exercise}

\begin{sol}
    If $g \cdot a = a$ for all $a \in G$, then $g = g1 = g \cdot 1 = 1$, so the kernel is $1$.
\end{sol}

\begin{exercise}{1.7.14}
    Let $G$ be a group and let $A = G$.
	Show that if $G$ is non-abelian then the maps defined by  
    \[
    	 g \cdot a = ag \quad \text{for all } g,a\in G 
    \]
    do \emph{not} satisfy the axioms of a (left) group action of $G$ on itself.
\end{exercise}

\begin{sol}
    Since $G$ is non-abelian, there exist $g, h \in G$ such that $gh \neq hg$.
    Then $g \cdot (h \cdot 1) = hg \neq gh = (gh) \cdot 1$, so the maps do not satisfy the axioms of a left group action.
\end{sol}

\begin{exercise}{1.7.15}
    Let $G$ be any group and let $A = G$.
	Show that the maps defined by  
    \[
    	 g \cdot a = ag^{-1} \quad \text{for all } g,a\in G 
    \]
    \emph{do} satisfy the axioms of a (left) group action of $G$ on itself.
\end{exercise}

\begin{sol}
    Let $g, h \in G$.
	Then for any $a \in A$, we have
    \[
    	g \cdot (h \cdot a) = g \cdot (ah\inv) = a(h\inv g\inv) = a(gh)\inv = (gh) \cdot a.
    \]
    Moreover, $1 \cdot a = a1 = a$, and thus this defines a left group action.
\end{sol}

\begin{exercise}!{1.7.16}
    Let $G$ be any group and let $A = G$.
	Show that the maps defined by  
    \[
    	 g \cdot a = gag^{-1} \quad \text{for all } g,a\in G 
    \]
    \emph{do} satisfy the axioms of a (left) group action.
    \note{This action of $G$ on itself is called \term{conjugation}.}
\end{exercise}

\begin{sol}
    Let $g, h \in G$.
	Then for any $a \in A$, we have
    \[
    	g \cdot (h \cdot a) = g \cdot (hah\inv) = g(hah\inv)g\inv = (gh)a(gh)\inv = (gh) \cdot a.
    \]
    Moreover, $1 \cdot a = 1a1\inv = a$, and therefore the maps define a left group action.
\end{sol}

\begin{exercise}{1.7.17}
    Let $G$ be a group and let $G$ act on itself by left conjugation, so each $g\in G$ maps $G$ to $G$ by
    \[
    	 x \mapsto gxg^{-1}. 
    \]
    For fixed $g\in G$, prove that conjugation by $g$ is an isomorphism from $G$ onto itself (i.e., is an automorphism of $G$---cf. \exref{1.6.20}).
	Deduce that $x$ and $gxg^{-1}$ have the same order for all $x\in G$ and that for any subset $A$ of $G$, $\abs{A} = \abs{gAg^{-1}}$ (here $gAg^{-1} = \set{gag^{-1} \mid a\in A}$).
\end{exercise}

\begin{sol}
    Let $\phi \colon x \mapsto gxg\inv$.
	Let $x, y \in G$.
	Then
    \[
    	\phi(x y) = g(xy)g\inv = (gxg\inv)(gyg\inv) = \phi(x)\phi(y),
    \]
    so $\phi$ is a homomorphism.
    Conjugation by $g\inv$ is a two-sided inverse of $\phi$, and hence $\phi$ is an isomorphism.
    By \exref{1.6.2}, we have $\abs{x} = \abs{gxg\inv}$.
	Moreover, the bijection $\phi$ restricts to a bijection from $A$ to $gAg\inv$, so $\abs{A} = \abs{gAg\inv}$.
\end{sol}

\begin{exercise}{1.7.18}
    Let $H$ be a group acting on a set $A$.
	Prove that the relation $\sim$ on $A$ defined by  
    \[
    	a \sim b \quad \text{if and only if} \quad a = hb \text{ for some } h \in H
    \]
    is an equivalence relation.
    \note{For each $x\in A$ the equivalence class of $x$ under $\sim$ is called the \term{orbit} of $x$ under the action of $H$.
	The orbits under the action of $H$ partition the set $A$.}
\end{exercise}

\begin{solitem}
    \item Let $a \in A$.
	Then $a = 1a$, hence $a \sim a$.
    \item Let $a, b \in A$ such that $a \sim b$.
	Then $a = hb$ for some $h \in H$, so $b = h\inv a$ and $b \sim a$.
    \item Let $a, b, c \in A$ such that $a \sim b$ and $b \sim c$.
	Then $a = h_1b$ and $b = h_2c$ for some $h_1, h_2 \in H$, so $a = h_1(h_2c) = (h_1h_2)c$ and $a \sim c$.
\end{solitem}

\begin{exercise}![Lagrange's Theorem]{1.7.19}
    Let $H$ be a subgroup of the finite group $G$ and let $H$ act on $G$ (here $A = G$) by left multiplication.
	Let $x\in G$ and let $\ocal$ be the orbit of $x$ under this action of $H$.
	Prove that the map  
    \[
    	H \to \ocal \quad \text{defined by} \quad h \mapsto hx
    \]
    is a bijection (hence all orbits have cardinality $\abs{H}$).
	From this and the preceding exercise deduce \term{Lagrange's Theorem}:  
    \begin{center}
        \emph{if $G$ is a finite group and $H$ is a subgroup of $G$ then $\abs{H}$ divides $\abs{G}$}.
    \end{center}
\end{exercise}

\begin{sol}
    Let $\phi \colon H \to \ocal$ be the map defined by $\phi(h) = hx$ for all $h \in H$.
    Then $\phi$ is surjective by definition of $\ocal$.
    Moreover, if $\phi(h_1) = \phi(h_2)$ for some $h_1, h_2 \in H$, then $h_1x = h_2x$, implying $h_1 = h_2$, so $\phi$ is injective.
    Therefore, $\phi$ is a bijection, whence $\abs{\ocal} = \abs{H}$.
    By \exref{1.7.18}, the orbits of the action of $H$ on $G$ partition $G$, each of cardinality $\abs{H}$.
    Thus, $\abs{G} = k \abs{H}$, where $k$ is the number of orbits and is finite since $G$ is finite, and hence $\abs{H}$ divides $\abs{G}$.
\end{sol}

\begin{exercise}{1.7.20}
    Show that the group of rigid motions of a tetrahedron is isomorphic to a subgroup (cf. \exref{1.1.26}) of $S_4$.
\end{exercise}

\begin{sol}
    Let $G$ be the group of rigid motions of a tetrahedron with vertices labeled $1$, $2$, $3$, and $4$.
	Then each $\alpha \in G$ sends vertices to vertices, so the elements of $G$ permute the set of vertices $A = \set{1, 2, 3, 4}$.
    This gives rise to the homomorphism
    \[
    	\phi \colon G \to S_4 \quad \text{where} \quad \phi(\alpha)(i) = \alpha(i) \quad \text{for all} \quad i \in A.
    \]
    To see that $\phi$ is injective, suppose $\phi(\alpha)$ is the identity permutation in $S_4$ for some $\alpha \in G$.
	Then for every vertex $i \in A$, we have $\phi(\alpha)(i) = i$, so that $\alpha(i) = i$.
	Since a rigid motion that fixes all vertices must be the identity rigid motion, $\alpha$ is the identity element of $G$.
	By \exref{1.6.14}, $\phi$ is injective, and \exref{1.6.13} gives $G \cong \phi(G)$, which is a subgroup of $S_4$.
\end{sol}

\begin{exercise}{1.7.21}
    Show that the group of rigid motions of a cube is isomorphic to $S_4$.\hint{This group acts on the set of four pairs of opposite vertices.}
\end{exercise}

\begin{sol}
	Let $G$ be the group of rigid motions of a cube, and let $A = \set{a_1, a_2, a_3, a_4}$ be the set of pairs of opposite vertices of the cube, where the cube is centered at the origin.
    By \exref{1.2.10}, we know that $\abs{G} = 24$.
    Each rigid motion $\alpha \in G$ permutes the set of pairs of opposite vertices, giving an action of $G$ on $A$.
    Thus, we have the homomorphism
    \[
    	\phi \colon G \to S_4 \quad \text{where} \quad \phi(\alpha)(a_i) = \alpha(a_i) \quad \text{for all} \quad a_i \in A.
    \]
    To see that $\phi$ is injective, suppose $\alpha \in \ker\phi$.
    Then $\phi(\alpha)(a_i) = a_i$ for all $a_i \in A$, so $\alpha(v) = \pm v$ for each vertex $v$ of the cube.
    Let $d(v, w)$ denote the distance between vertices $v$ and $w$.
    For some vertex $v$, suppose $\alpha(v) = v$, but $\alpha(w) = -w$ for an adjacent vertex $w$.
    Then $d(\alpha(v), \alpha(w)) = d(v, -w) \neq d(v, w)$, contradicting that $\alpha$ is a rigid motion.
    Since any two vertices are joined by some path of edges, $\alpha$ must send all vertices to themselves or swap each vertex with its opposite vertex.
    The former case corresponds to the identity rigid motion, while the latter case corresponds to a reflection through the origin, reversing the orientation, which is not a rigid motion of the cube.
    Hence, the kernel of $\phi$ is trivial, and $\phi$ is injective by \exref{1.6.14}.
    Since $\abs{G} = 24 = \abs{S_4}$, it follows that $\phi$ is surjective as well, and thus $G \cong S_4$.
\end{sol}

\begin{exercise}{1.7.22}
    Show that the group of rigid motions of an octahedron is isomorphic to a subgroup (cf. \exref{1.1.26})of $S_4$.  \hint{This group acts on the set of four pairs of opposite faces.}
    Deduce that the groups of rigid motions of a cube and of an octahedron are isomorphic.
    \note{These groups are isomorphic because these solids are ``dual''---see \emph{Introduction to Geometry} by H. Coxeter, Wiley, 1961.
    We shall see later that the groups of rigid motions of the dodecahedron and icosahedron are isomorphic as well---these solids are also dual.}
\end{exercise}

\begin{sol}
    Let $G$ be the group of rigid motions of an octahedron and let $A$ be the set of four pairs of opposite faces of the octahedron.
	Each $\alpha \in G$ sends each pair of opposite faces to a pair of opposite faces, so $G$ acts on $A$.
    By the argument in \exref{1.7.21} with centers of faces in place of vertices and pairs of opposite faces in place of pairs of opposite vertices, we see that the homomorphism from $G$ to $S_4$ induced by this action is injective.
    By \exref{1.2.11}, we know that $\abs{G} = 24 = \abs{S_4}$, so the injective homomorphism is also surjective, wherefore it is an isomorphism.
    By \exref{1.7.21}, the group of rigid motions of a cube is isomorphic to $S_4 \cong G$.
\end{sol}

\begin{exercise}{1.7.23}
    Explain why the action of the group of rigid motions of a cube on the set of three pairs of opposite faces is not faithful.
	Find the kernel of this action.
\end{exercise}

\begin{sol}
    Let $G$ be the group of rigid motions of a cube.
    By \exref{1.2.10}, $\abs{G} = 24 > 6 = \abs{S_3}$, so the permutation representation cannot be injective, and hence the action is not faithful.
    Let $\phi$ be the permutation representation of $G$ acting on the set of three pairs of opposite faces and let $K = \ker\phi$.
    By \exref{1.6.14}, $K$ is a subgroup of $G$.
    Now let $K$ act on $G$ by left multiplication.
    Let $\ocal_a$ be the orbit of some $a \in G$ under this action.
    By \exref{1.7.18}, the orbits partition $G$, and by \exref{1.7.19}, each orbit has size $\abs{K}$.

    Place the cube with vertices $(\pm 1, \pm 1, \pm 1)$, let $r_1$ be the rotation by $90^\circ$ about the $z$-axis, and let $r_2$ be the map $(x, y, z) \mapsto (z, x, y)$ defined by a $120^\circ$ rotation about the axis through the vertices $(1, 1, 1)$ and $(-1, -1, -1)$.
    Observe that $r_1$ swaps the $x$- and $y$-pairs of opposite faces of the cube, while $r_2$ sends the $x$-pair of opposite faces to the $y$-pair, the $y$-pair to the $z$-pair, and the $z$-pair to the $x$-pair.
    Denoting the $x$-, $y$-, and $z$-pairs of opposite faces by $1$, $2$, and $3$, respectively, we have $\phi(r_1) = \cyc{1 2}$ and $\phi(r_2) = \cyc{1 2 3}$.
    Since $\phi(G)$ is closed under composition, it must contain the subgroup generated by $\cyc{1 2}$ and $\cyc{1 2 3}$, which is all of $S_3$ by \exref{1.3.20}.
    Therefore, $\phi$ is surjective.
    For any $a, b \in G$, we have $b \in \ocal_a$ if and only if $ba\inv \in K$, which holds if and only if $\phi(b) = \phi(a)$, so the orbits of $G$ under the action of $K$ correspond to the fibers of $\phi$.
    Therefore, there are $\abs{S_3} = 6$ distinct orbits of $G$ under the action of $K$.
    Since $\abs{G} = 24$ and each orbit has size $\abs{K}$, we have $24 = \abs{G} = \abs{K} \cdot 6$, so $\abs{K} = 4$.
    Finally, the identity and the three $180^\circ$ rotations about the coordinate axes are in $K$, and thus all elements of $K$ are accounted for.
\end{sol}
