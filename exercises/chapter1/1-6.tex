\subsection{Homomorphisms and Isomorphisms}

Let $G$ and $H$ be groups.

\begin{exercise}!{1.6.1}
    Let $\phi \colon G \to H$ be a homomorphism.
    \begin{subexercises}
        \item Prove that $\phi(x^n) = \phi(x)^n$ for all $n \in \zp$.
        \item Do part (a) for $n = - 1$ and deduce that $\phi(x^n) = \phi(x)^n$ for all $n \in \zbb$.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Fix $x \in G$.
    We proceed by induction on $n$, where the base case $n = 1$ is trivial.
	Suppose the result holds for some $n \in \zp$, so $\phi(x^n) = \phi(x)^n$.
    We then have
    \[
    	\phi(x^{n + 1}) = \phi(x^nx) = \phi(x^n)\phi(x) = \phi(x)^n\phi(x) = \phi(x)^{n + 1}.
    \]
    \item Let $1_G$ and $1_H$ denote the identity elements of $G$ and $H$, respectively.
    We have $\phi(1_G) = \phi(1_G1_G) = \phi(1_G)^2$, so Proposition 1.2 gives $\phi(1_G) = 1_H$.
    Thus, for any $x \in G$, we have
    \[
        1_H = \phi(xx\inv) = \phi(x)\phi(x\inv),
    \]
    and left multiplying by $\phi(x)\inv$ gives $\phi(x)\inv = \phi(x\inv)$.
    For $n = 0$, we have $\phi(x^0) = \phi(1_G) = 1_H = \phi(x)^0$.
    Then for any $n < 0$, we use part (a) with $-n$ to yield
    \[
        \phi(x^n) = \phi((x\inv)^{-n}) = \phi(x\inv)^{-n} = \phi(x)^n. \qedhere
    \]
\end{solenum}

\begin{exercise}!{1.6.2}
    If $\phi \colon G \to H$ is an isomorphism, prove that $\abs{\phi(x)} = \abs{x}$ for all $x \in G$.
	Deduce that any two isomorphic groups have the same number of elements of order $n$ for each $n \in \zbb^+$.
	Is the result true if $\phi$ is only assumed to be a homomorphism?
\end{exercise}

\begin{sol}
    Let $x \in G$ and $k \in \zp$.
    By \exref{1.6.1} and injectivity of $\phi$,
    \[
        x^k = 1_G \iff \phi(x)^k = \phi(x^k) = 1_H.
    \]
    Then $x$ and $\phi(x)$ have the same set of positive integers $k$ such that $x^k = 1_G$ and $\phi(x)^k = 1_H$, respectively, and thus $\abs{x} = \abs{\phi(x)}$.
    Consequently, the restriction $\phi|_S$, where $S = \set{g \in G \mid \abs{g} = n}$, is a bijection onto $\set{h \in H \mid \abs{h} = n}$, so the number of elements of order $n$ in $G$ and $H$ is the same.

    The result is not true if $\phi$ is only assumed to be a homomorphism.
    Consider $\phi \colon G \to 1$ for some nontrivial group $G$.
    Then any nonidentity $x \in G$ has order greater than 1, but $\phi(x) = 1$ has order 1.
\end{sol}

\begin{exercise}{1.6.3}
    If $\phi \colon G \to H$ is an isomorphism, prove that $G$ is abelian if and only if $H$ is abelian.
	If $\phi \colon G \to H$ is a homomorphism, what additional conditions on $\phi$ (if any) are sufficient to ensure that if $G$ is abelian, then so is $H$?
\end{exercise}

\begin{sol}
    \leavevmode

    \rightimp Suppose that $G$ is abelian.
    Given $h_1, h_2 \in H$, there exist $g_1, g_2 \in G$ such that $\phi(g_1) = h_1$ and $\phi(g_2) = h_2$.
    Then
    \[
        h_1h_2 = \phi(g_1)\phi(g_2) = \phi(g_1g_2) = \phi(g_2g_1) = \phi(g_2)\phi(g_1) = h_2h_1,
    \]
    so that $H$ is abelian.

    \leftimp Suppose that $H$ is abelian.
    Then for any $g_1, g_2 \in G$, we have
    \[
        \phi(g_1g_2) = \phi(g_1)\phi(g_2) = \phi(g_2)\phi(g_1) = \phi(g_2g_1),
    \]
    and injectivity of $\phi$ implies $g_1g_2 = g_2g_1$, so $G$ is abelian.
    
    Let $\phi \colon G \to H$ be a homomorphism.
	In the \rightimp direction, observe that the only additional property of $\phi$ used was its surjectivity, so it is sufficient for $\phi$ to be surjective.
\end{sol}

\begin{exercise}{1.6.4}
    Prove that the multiplicative groups $\rbb - \set{0}$ and $\cbb- \set{0}$ are not isomorphic.
\end{exercise}

\begin{sol}
    By \exref{1.6.2}, isomorphisms preserve the number of elements of each order.
    The element $i \in \cbb - \set{0}$ has order 4, and if $x \in \rbb - \set{0}$ is such that $x^4 = 1$, then $x = \pm 1$, which has order 1 or 2.
\end{sol}

\begin{exercise}{1.6.5}
    Prove that the additive groups $\rbb$ and $\qbb$ are not isomorphic.
\end{exercise}

\begin{sol}
    $\rbb$ is uncountable, while $\qbb$ is countable, so no such isomorphism can exist.
\end{sol}

\begin{exercise}{1.6.6}
    Prove that the additive groups $\zbb$ and $\qbb$ are not isomorphic.

    \remark{In fact, no nontrivial homomorphism exists from $\qbb$ to $\zbb$.}
\end{exercise}

\begin{sol}
    Suppose that $\phi \colon \qbb \to \zbb$ is an isomorphism.
    Because $\phi(0) = 0$ by \exref{1.6.1}, injectivity of $\phi$ gives $r = \phi(1) \neq 0$.
    Fix $n > \abs{r}$.
    By \exref{1.6.1}(a) in additive notation, we have
    \[
        r = \phi(1) = \phi\left(n \cdot \frac{1}{n}\right) = n \phi\left(\frac{1}{n}\right),
    \]
    and thus $\phi(\frac{1}{n}) = \frac{r}{n} \in \zbb$, contradicting the choice of $n$.
\end{sol}

\begin{exercise}{1.6.8}
    Prove that if $n \neq m$, $S_n$ and $S_m$ are not isomorphic.
\end{exercise}

\begin{sol}
    If $n \neq m$, then $n! \neq m!$ so that $\abs{S_n} \neq \abs{S_m}$.
\end{sol}

\begin{exercise}{1.6.10}
    Fill in the details of the proof that the symmetric groups $S_\Delta$ and $S_\Omega$ are isomorphic if $\abs{\Delta} = \abs{\Omega}$ as follows: let $\theta \colon \Delta \to \Omega$ be a bijection.
	Define
    \[
        \phi \colon S_\Delta \to S_\Omega \quad \text{by} \quad \phi(\sigma) = \theta \circ \sigma \circ \theta\inv \quad \text{for all } \sigma \in S_\Delta
    \]
    and prove the following:
    \begin{subexercises}
        \item $\phi$ is well defined, that is, if $\sigma$ is a permutation of $\Delta$ then $\theta \circ \sigma \circ \theta\inv$ is a permutation of $\Omega$.
        \item $\phi$ is a bijection from $S_\Delta$ onto $S_\Omega$.
        \hint{Find a $2$-sided inverse for $\phi$.}
        \item $\phi$ is a homomorphism, that is, $\phi(\sigma \circ \tau) = \phi(\sigma) \circ \phi(\tau)$.
    \end{subexercises}
    Note the similarity to the \emph{change of basis} or \emph{similarity} transformations for matrices (we shall see the connections between these later in the text).
\end{exercise}

\begin{solenum}
    \item The map $\theta \circ \sigma \circ \theta\inv$ is a composition of bijections that takes $\Omega \to \Delta \to \Delta \to \Omega$, and hence is a permutation of $\Omega$, so $\phi$ is well defined.
    \item Define the mapping 
    \[
    	\psi \colon S_\Omega \to S_\Delta\quad \text{by} \quad \psi(\tau) = \theta\inv \circ \tau \circ \theta \quad \text{for all } \tau \in S_\Omega.
    \]
    The map $\theta\inv \circ \tau \circ \theta$ is a composition of bijections that takes $\Delta \to \Omega \to \Omega \to \Delta$, and hence is a permutation of $\Delta$, so $\psi$ is well defined.
	For any $\sigma \in S_\Delta$, we have
    \[
    	\psi(\phi(\sigma)) = \theta\inv \circ (\theta \circ \sigma \circ \theta\inv) \circ \theta = 1_\Delta \circ \sigma \circ 1_\Delta = \sigma,
    \]
    and for any $\tau \in S_\Omega$, we have
    \[
    	\phi(\psi(\tau)) = \theta \circ (\theta\inv \circ \tau \circ \theta) \circ \theta\inv = 1_\Omega \circ \tau \circ 1_\Omega = \tau,
    \]
    wherefore $\psi$ is a two-sided inverse for $\phi$.
	Hence, $\phi$ is a bijection from $S_\Delta$ onto $S_\Omega$.
    \item Let $\sigma, \tau \in S_\Delta$.
	Then
    \[
    	\phi(\sigma \circ \tau) = \theta \circ (\sigma \circ \tau) \circ \theta\inv = (\theta \circ \sigma \circ \theta\inv) \circ (\theta \circ \tau \circ \theta\inv) = \phi(\sigma) \circ \phi(\tau).
    \]
    Therefore, $\phi$ is a homomorphism, whence it is an isomorphism, so $S_\Delta \cong S_\Omega$.
\end{solenum}

\begin{exercise}{1.6.11}
    Let $A$ and $B$ be groups.
	Prove that $A \times B \cong B \times A$.
\end{exercise}

\begin{sol}
    Consider the map $\phi \colon A \times B \to B \times A$ given by
    \[
    	\phi((a, b)) = (b, a) \quad \text{for all } a \in A, b \in B.
    \]
    The map $(b, a) \mapsto (a, b)$ is the two-sided inverse of $\phi$, so $\phi$ is bijective.
	Moreover, for all $(a_1, b_1), (a_2, b_2) \in A \times B$, we have
    \[
    	\phi((a_1, b_1)(a_2, b_2)) = \phi((a_1a_2, b_1b_2)) = (b_1b_2, a_1a_2) = (b_1, a_1)(b_2, a_2) = \phi((a_1, b_1))\phi((a_2, b_2)).
    \]
    Therefore, $\phi$ is a homomorphism, and hence an isomorphism, so $A \times B \cong B \times A$.
\end{sol}

\begin{exercise}{1.6.12}
    Let $A$, $B$, and $C$ be groups and let $G = A \times B$ and $H = B \times C$.
	Prove that $G \times C \cong A \times H$.
\end{exercise}

\begin{sol}
    Consider the map $\phi \colon G \times C \to A \times H$ given by
    \[
    	\phi(((a, b), c)) = (a, (b, c)) \quad \text{for all } a \in A, b \in B, c \in C.
    \]
    The map $(a, (b, c)) \mapsto ((a, b), c)$ is the two-sided inverse of $\phi$, so $\phi$ is bijective.
	As in \exref{1.6.11}, $\phi$ is a homomorphism, and hence an isomorphism, so $G \times C \cong A \times H$.
\end{sol}

\begin{exercise}![Image]{1.6.13}
    Let $G$ and $H$ be groups and let $\phi \colon G \to H$ be a homomorphism.
	Prove that the \term{image} of $\phi$, $\phi(G)$, is a subgroup of $H$ (cf. \exref{1.1.26}).
	Prove that if $\phi$ is injective then $G \cong \phi(G)$.
\end{exercise}

\begin{sol}
    Let $g_1, g_2 \in G$.
    By \exref{1.6.1}(b), $\phi(1_G) = 1_H$, so $1_H \in \phi(G)$, and
    \[
        \phi(g_1g_2) = \phi(g_1)\phi(g_2) \in \phi(G), \quad \phi(g_1\inv) = \phi(g_1)\inv \in \phi(G),
    \]
    showing that $\phi(G)$ is a subgroup of $H$.

    Now suppose $\phi$ is injective.
    The map $\psi \colon G \to \phi(G)$ defined by $g \mapsto \phi(g)$ is injective since $\phi$ is injective, surjective by definition of $\phi(G)$, and a homomorphism since $\phi$ is a homomorphism, so $G \cong \phi(G)$.
\end{sol}

\begin{exercise}![Kernel]{1.6.14}
    Let $G$ and $H$ be groups and let $\phi \colon G \to H$ be a homomorphism.
	Define the \term{kernel} of $\phi$ to be $\set{g \in G \mid \phi(g) = 1_H }$ (so the kernel is the set of elements in $G$ which map to the identity of $H$, i.e., is the fiber over the identity of $H$).
	Prove that the kernel of $\phi$ is a subgroup (cf. \exref{1.1.26}) of $G$.
	Prove that $\phi$ is injective if and only if the kernel of $\phi$ is the identity subgroup of $G$.
\end{exercise}

\begin{sol}
    Denote the kernel of $\phi$ by $\ker\phi$.
	By \exref{1.6.1}(b), $\phi(1_G) = 1_H$, so $1_G \in \ker\phi$, and for $g_1, g_2 \in \ker\phi$, we have
    \[
        \phi(g_1g_2) = \phi(g_1)\phi(g_2) = 1_H \cdot 1_H = 1_H, \quad \phi(g_1\inv) = \phi(g_1)\inv = 1_H\inv = 1_H,
    \]
    so $g_1g_2, g_1\inv \in \ker\phi$, and thus $\ker\phi$ is a subgroup of $G$.

    \rightimp Suppose $\phi$ is injective, and let $g \in \ker\phi$.
    By definition, $\phi(g) = 1_H = \phi(1_G)$, and injectivity of $\phi$ gives $g = 1_G$, so $\ker\phi = 1$.

    \leftimp Suppose $\ker\phi = 1$, and let $g_1, g_2 \in G$ such that $\phi(g_1) = \phi(g_2)$.
    By \exref{1.6.1}(b), we have
    \[
        \phi(g_1g_2\inv) = \phi(g_1)\phi(g_2\inv) = \phi(g_1)\phi(g_2)\inv = 1_H,
    \]
    so $g_1g_2\inv \in \ker\phi$.
    Then $g_1g_2\inv = 1_G$, so $g_1 = g_2$, and hence $\phi$ is injective.
\end{sol}

\begin{exercise}{1.6.15}
    Define a map $\pi \colon \rbb^2 \to \rbb$ by $\pi((x,y)) = x$.
	Prove that $\pi$ is a homomorphism and find the kernel of $\pi$ (cf. \exref{1.6.14}).
\end{exercise}

\begin{sol}
    Let $(x_1, y_1), (x_2, y_2) \in \rbb^2$.
	Then
    \[
    	\pi((x_1, y_1) + (x_2, y_2)) = \pi((x_1 + x_2, y_1 + y_2)) = x_1 + x_2 = \pi((x_1, y_1)) + \pi((x_2, y_2)),
    \]
    so $\pi$ is a homomorphism.
    Moreover, $(x, y) \in \ker\pi$ if and only if $\pi((x, y)) = x = 0$, so
    \[
        \ker\pi = \set{(0, y) \mid y \in \rbb}. \qedhere
    \]
\end{sol}

\begin{exercise}{1.6.16}
    Let $A$ and $B$ be groups and let $G$ be their direct product, $A \times B$.
	Prove that the maps $\pi_1 \colon G \to A$ and $\pi_2 \colon G \to B$ defined by $\pi_1((a,b)) = a$ and $\pi_2((a,b)) = b$ are homomorphisms and find their kernels (cf. \exref{1.6.14}).
\end{exercise}

\begin{sol}
    As in \exref{1.6.15}, $\pi_1$ and $\pi_2$ are homomorphisms.
	Moreover, $(a, b) \in \ker\pi_1$ if and only if $\pi_1((a, b)) = a = 1_A$, and $(a, b) \in \ker\pi_2$ if and only if $\pi_2((a, b)) = b = 1_B$.
    Therefore,
    \[
        \ker\pi_1 = \set{(1_A, b) \mid b \in B}, \quad \ker\pi_2 = \set{(a, 1_B) \mid a \in A}. \qedhere
    \]
\end{sol}

\begin{exercise}{1.6.17}
    Let $G$ be any group.
	Prove that the map from $G$ to itself defined by $g \mapsto g\inv$ is a homomorphism if and only if $G$ is abelian.
\end{exercise}

\begin{sol}
    Let $\phi \colon G \to G$ be defined by $\phi(g) = g\inv$.

    \rightimp Suppose $\phi$ is a homomorphism.
	Then for $g, h \in G$, we have
    \[
        gh = (h\inv g\inv)\inv = \phi(h\inv g\inv) = \phi(h\inv)\phi(g\inv) = hg,
    \]
    so $G$ is abelian.
    
    \leftimp Suppose $G$ is abelian.
	Then for $g, h \in G$, we have
    \[
    	\phi(gh) = (gh)\inv = h\inv g\inv = g\inv h\inv = \phi(g)\phi(h),
    \]
    so that $\phi$ is a homomorphism.
\end{sol}

\begin{exercise}{1.6.18}
    Let $G$ be any group.
	Prove that the map from $G$ to itself defined by $g \mapsto g^2$ is a homomorphism if and only if $G$ is abelian.
\end{exercise}

\begin{sol}
    Let $\phi \colon G \to G$ be defined by $\phi(g) = g^2$.
    Then for all $g, h \in G$, we have
    \[
        \phi(gh) = \phi(g)\phi(h) \iff (gh)^2 = g^2 h^2 \iff hg = gh. \qedhere
    \]
\end{sol}

\begin{exercise}{1.6.19}
    Let $G = \set{ z \in \cbb\mid z^n = 1 \text{ for some } n \in \zbb^+ }$.
	Prove that for any fixed integer $k > 1$ the map from $G$ to itself defined by $z \mapsto z^k$ is a surjective homomorphism but is not an isomorphism.
\end{exercise}

\begin{sol}
    Define the map $\phi \colon G \to G$ by $\phi(z) = z^k$.
    Since $G \subseteq \cbb$, $G$ is abelian, so for any $z, w \in G$,
    \[
    	\phi(zw) = (zw)^k = z^kw^k = \phi(z)\phi(w),
    \]
    and hence $\phi$ is a homomorphism.
	To show surjectivity, let $w \in G$ with $w^n = 1$ for some $n \in \zp$.
    Then there exists $z \in \cbb$ such that $z^k = w$, so $z^{nk} = (z^k)^n = w^n = 1$ and hence $z \in G$.
    Moreover, $\phi(z) = z^k = w$, so $\phi$ is surjective.
    Finally, $u = e^{2\pi i/k} \in G$ is such that $\phi(u) = u^k = 1$ with $u \neq 1$ since $k > 1$, so $\phi$ is not injective, and therefore not an isomorphism.
\end{sol}

\begin{exercise}![The Automorphism Group]{1.6.20}
    Let $G$ be a group and let $\aut(G)$ be the set of all isomorphisms from $G$ onto $G$.
	Prove that $\aut(G)$ is a group under function composition (called the \term{automorphism group} of $G$ and the elements of $\aut(G)$ are called \term{automorphisms} of $G$).
\end{exercise}

\begin{sol}
    Recall that function composition is associative.
    Observe that $\aut(G)$ contains the identity map $1 \colon G \to G$ defined by $1(g) = g$ for all $g \in G$.
    For any $\phi, \psi \in \aut(G)$ and $g, h \in G$, we have
    \[
        (\phi \circ \psi)(gh) = \phi(\psi(gh)) = \phi(\psi(g)\psi(h)) = \phi(\psi(g))\phi(\psi(h)) = (\phi \circ \psi)(g)(\phi \circ \psi)(h),
    \]
    so $\phi \circ \psi$ is a bijective homomorphism, and hence $\aut(G)$ is closed under composition.
    Finally, for any $\phi \in \aut(G)$, $\phi\inv$ exists and is also a bijection. 
    Then for any $g, h \in G$, we have
    \[
        \phi\inv(gh) = \phi\inv(\phi(\phi\inv(g))\phi(\phi\inv(h))) = \phi\inv(\phi(\phi\inv(g)\phi\inv(h))) = \phi\inv(g)\phi\inv(h),
    \]
    so $\phi\inv$ is a homomorphism, and $\phi\inv \in \aut(G)$.
    Therefore, $(\aut(G), \circ)$ is a group.
\end{sol}

\begin{exercise}{1.6.21}
    Prove that for each fixed nonzero $k \in \qbb$ the map from $\qbb$ to itself defined by $q \mapsto kq$ is an automorphism of $\qbb$ (cf. \exref{1.6.20}).
\end{exercise}

\begin{sol}
    Let $\phi \colon \qbb \to \qbb$ be defined by $\phi(q) = kq$ for all $q \in \qbb$.
	For any $p, q \in \qbb$, we have
    \[
    	\phi(p + q) = k(p + q) = kp + kq = \phi(p) + \phi(q),
    \]
    so that $\phi$ is a homomorphism.
	The map $q \mapsto \frac{q}{k}$ is the two-sided inverse of $\phi$, so $\phi \in \aut(\qbb)$.
\end{sol}

\begin{exercise}{1.6.22}
    Let $A$ be an abelian group and fix some $k \in \zbb$.
	Prove that the map $a \mapsto a^k$ is a homomorphism from $A$ to itself.
	If $k = -1$ prove that this homomorphism is an isomorphism (i.e., is an automorphism of $A$).
\end{exercise}

\begin{sol}
    Let $\phi \colon A \to A$ be defined by $\phi(a) = a^k$ for all $a \in A$.
	For any $a_1, a_2 \in A$, we have
    \[
    	\phi(a_1a_2) = (a_1a_2)^k = a_1^k a_2^k = \phi(a_1)\phi(a_2),
    \]
    and thus $\phi$ is a homomorphism.
    If $k = -1$, then $\phi \circ \phi = 1$, so $\phi = \phi\inv$, and hence $\phi \in \aut(A)$.
\end{sol}

\begin{exercise}!{1.6.23}
    Let $G$ be a finite group which possesses an automorphism $\sigma$ (cf. \exref{1.6.20}) such that $\sigma(g) = g$ if and only if $g = 1$.
	If $\sigma^2$ is the identity map from $G$ to $G$, prove that $G$ is abelian (such an automorphism $\sigma$ is called \term{fixed point free} of order 2).
\hint{Show that every element of $G$ can be written in the form $x\inv\sigma(x)$ and apply $\sigma$ to such an expression.}
\end{exercise}

\begin{sol}
    Define the map $\phi \colon G \to G$ by
    \[
        \phi(x) = x\inv \sigma(x) \quad \text{for all } x \in G.
    \]
	If $\phi(x) = \phi(y)$ for $x, y \in G$, then $x\inv \sigma(x) = y\inv \sigma(y)$, and hence $yx\inv = \sigma(yx\inv)$.
	Since $\sigma$ is fixed point free, $yx\inv = 1$, so $y = x$, and thus $\phi$ is injective.
	Since $G$ is finite, $\phi$ is also surjective, and every element of $G$ may be written in the form $x\inv \sigma(x)$ for some $x \in G$.

    For any $g \in G$, there exists $x \in G$ such that $g = x\inv \sigma(x)$.
	Applying $\sigma$ to both sides, we have
    \[
    	\sigma(g) = \sigma(x\inv \sigma(x)) = \sigma(x\inv) \sigma^2(x) = \sigma(x)\inv x = (x\inv \sigma(x))\inv = g\inv.
    \]
    By \exref{1.6.17}, $G$ is abelian.
\end{sol}

\begin{exercise}!{1.6.24}
    Let $G$ be a finite group and let $x$ and $y$ be distinct elements of order 2 in $G$ that generate $G$.
	Prove that $G \cong D_{2n}$, where $n = \abs{xy}$.
    \hint{See \exref{1.2.6}.}
\end{exercise}

\begin{sol}
    Observe that $G$ must contain the distinct elements $1$, $x$, and $y$.
    Put $t = xy$, and note that $x \neq y$ gives $t \neq 1$, so $n \geq 2$.
    Note that $t^n = (xy)^n = 1$, $x^2 = 1$, and $txt = (xy)x(xy) = xyy = x$, so $tx = xt\inv$.
    Moreover, $y = xt$, so $G = \gen{x, t}$.
    Then the map $r \mapsto t$ and $s \mapsto x$ extends to a surjective homomorphism $\phi \colon D_{2n} \to G$.

    Now, \exref{1.1.32} gives that $1, t, \ldots, t^{n-1}$ are all distinct elements of $G$.
    If $xt^j = xt^k$ for some $0 \leq j, k \leq n-1$ such that $j \neq k$, then left cancelling by $x$ gives $t^j = t^k$, contradicting the distinctness of $1, t, \ldots, t^{n-1}$, so the elements $x, xt, \ldots, xt^{n-1}$ are also distinct.
    If $x \in \gen{t}$, then $x$ must commute with $t$, so $xt = tx = xt\inv$.
    This yields $t^2 = 1$, so $y = xt \in \gen{t}$ as well, and therefore $G = \gen{t}$.
    But then $G$ would have at most 2 elements, contradicting the fact that $1$, $x$, and $y$ are distinct elements of $G$.
    Therefore, no $xt^j \in \gen{t}$, for otherwise $x \in \gen{t}$, so all the elements $1, t, \ldots, t^{n-1}, x, xt, \ldots, xt^{n-1}$ are distinct.
    Thus, $\abs{G} = 2n$ and $G \cong D_{2n}$.
\end{sol}

\begin{exercise}{1.6.25}
    Let $n \in \zbb^+$, let $r$ and $s$ be the usual generators of $D_{2n}$ and let $\theta = 2\pi / n$.
    \begin{subexercises}
        \item Prove that the matrix 
        \[
            \begin{pmatrix}
                \cos \theta & -\sin \theta \\
                \sin \theta & \cos \theta
            \end{pmatrix}
        \]
        is the matrix of the linear transformation which rotates the $x,y$ plane about the origin in a counterclockwise direction by $\theta$ radians.
        \item Prove that the map $\phi \colon D_{2n} \to \gl_2(\rbb)$ defined on generators by
        \[
            \phi(r) = 
            \begin{pmatrix}
                \cos \theta & -\sin \theta \\
                \sin \theta & \cos \theta
            \end{pmatrix}
            \quad \text{and} \quad
            \phi(s) = 
            \begin{pmatrix}
                0 & 1 \\
                1 & 0
            \end{pmatrix}
        \]
        extends to a homomorphism of $D_{2n}$ into $\gl_2(\rbb)$.
        \item Prove that the homomorphism $\phi$ in part (b) is injective.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Let $\rbb^2$ be spanned by the standard basis vectors $\mb e_1 = (1 \quad 0)^t$ and $\mb e_2 = (0 \quad 1)^t$ and let $T$ denote the linear transformation which rotates the $x,y$ plane about the origin in a counterclockwise direction by $\theta$ radians.
	Then
    \[
        T(\mb e_1) = 
        \begin{pmatrix}
            \cos \theta \\
            \sin \theta
        \end{pmatrix} \quad \text{and} \quad
        T(\mb e_2) = 
        \begin{pmatrix}
            -\sin \theta \\
            \cos \theta
        \end{pmatrix},
    \]
    so the matrix of $T$ with respect to $\mb e_1$ and $\mb e_2$ is exactly the desired rotation matrix.
    \item By induction on $k$, $\phi(r)^k$ is given by
    \[
        \phi(r)^k = 
        \begin{pmatrix}
            \cos (k\theta) & -\sin (k\theta) \\
            \sin (k\theta) & \cos (k\theta)
        \end{pmatrix},
    \]
    and thus $\phi(r)^n = I_2$.
    Since $\det\phi(r) = 1$, we have
    \[
        \phi(r)\inv = 
        \begin{pmatrix}
            \cos \theta & \sin \theta \\
            -\sin \theta & \cos \theta
        \end{pmatrix}.
    \]
	Direct computation gives $\phi(s)^2 = I_2$, and
    \[
        \phi(r)\phi(s) = 
        \begin{pmatrix}
            \cos \theta & -\sin \theta \\
            \sin \theta & \cos \theta
        \end{pmatrix}
        \begin{pmatrix}
            0 & 1 \\
            1 & 0
        \end{pmatrix} =
        \begin{pmatrix}
            -\sin\theta & \cos\theta \\
            \cos\theta & \sin\theta
        \end{pmatrix} = 
        \begin{pmatrix}
            0 & 1 \\
            1 & 0
        \end{pmatrix}
        \begin{pmatrix}
            \cos\theta & \sin\theta \\
            -\sin\theta & \cos\theta
        \end{pmatrix} = 
        \phi(s)\phi(r)\inv.
    \]
	Since these relations match those of $D_{2n}$, $\phi$ extends to a homomorphism from $D_{2n}$ to $\gl_2(\rbb)$.
    \item By \exref{1.6.14}, it suffices to show $\ker\phi = 1$.
    Let $g \in \ker\phi$, so $\phi(g) = I_2$.
    If $g = r^k$ for some $1 \leq k < n$, then $\phi(g)_{11} = \cos(k\theta) = \cos(\frac{2\pi k}{n}) \neq 1$, so $\phi(g) \neq I_2$.
    If $g = sr^k$ for some $0 \leq k < n$, then $\phi(g) = \phi(s)\phi(r)^k$.
    Since $\det\phi(s) = -1$ and $\det\phi(r)^k = 1$, it follows that $\det\phi(g) = - 1 \neq 1 = \det I_2$.
    Hence, $g = 1$, and $\ker\phi = 1$, so $\phi$ is injective.
\end{solenum}

\begin{exercise}{1.6.26}
    Let $i$ and $j$ be the generators of $Q_8$ described in Section 5.
	Prove that the map $\phi$ from $Q_8$ to $\gl_2(\cbb)$ defined on generators by
    \[
        \phi(i) = 
        \begin{pmatrix}
            \sqrt{-1} & 0 \\
            0 & -\sqrt{-1}
        \end{pmatrix}
        \quad \text{and} \quad
        \phi(j) = 
        \begin{pmatrix}
            0 & -1 \\
            1 & 0
        \end{pmatrix}
    \]
    extends to a homomorphism.
	Prove that $\phi$ is injective.
\end{exercise}

\begin{sol}
    Matrix multiplication shows that the relations described in \exref{1.5.3} are satisfied, so that $\phi$ extends to a homomorphism from $Q_8$ to $\gl_2(\cbb)$.
	By \exref{1.6.14}, it suffices to show that $\ker\phi = 1$.
    By \exref{1.5.1}, every element of $Q_8 - \set{\pm 1}$ squares to $-1$, so a nontrivial kernel must contain $-1$.
    However, $\phi(-1) = -I_2 \neq I_2$, so $-1 \notin \ker\phi$.
    Therefore, $\ker\phi = 1$, and $\phi$ is injective.
\end{sol}
