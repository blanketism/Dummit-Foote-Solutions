\subsection{Dihedral Groups}

In these exercises, $D_{2n}$ has the usual presentation 
$D_{2n} = \gen{r, s \mid r^n = s^2 = 1, rs = sr\inv}$.

\begin{exercise}{1.2.1}
    Compute the order of each of the elements in the following groups:

    (a) $D_6$ \quad (b) $D_8$ \quad (c) $D_{10}.$
\end{exercise}

\begin{sol}
    For any reflection $sr^k \in D_{2n}$, it is clear that $sr^k \neq 1$.
	By the properties of $D_{2n}$:
    \[
    	(sr^k)^2 = s(r^ks)r^k = s(sr^{-k})r^k = r^{-k}r^k = 1
    \]
    so that every reflection has order 2.
	We now compute the orders of the rotations in each group.
    \begin{subexercises}
        \item $D_6 = \set{1, r, r^2, s, sr, sr^2}$.
	    Then $\abs{1} = 1$ and $\abs{r} = \abs{r^2} = 3$.
        \item $D_8 = \set{1, r, r^2, r^3, s, sr, sr^2, sr^3}$.
	    Then $\abs{1} = 1$, $\abs{r^2} = 2$, and $\abs{r} = \abs{r^3} = 4$.
        \item $D_{10} = \set{1, r, r^2, r^3, r^4, s, sr, sr^2, sr^3, sr^4}$.
	    Then $\abs{1} = 1$ and $\abs{r} = \abs{r^2} = \abs{r^3} = \abs{r^4} = 5$. \qedhere
    \end{subexercises}
\end{sol}

\begin{exercise}{1.2.2}
    Use the generators and relations above to show that if $x$ is any element of $D_{2n}$ which is not a power of $r$, then $rx = xr\inv$.
\end{exercise}

\begin{sol}
    If $x$ is not a power of $r$, then $x = sr^k$ for some $0 \leq k < n$.
	Therefore, we have
    \[
    	rx = (rs)r^k = s r\inv r^k = (s r^k) r\inv = x r\inv. \qedhere
    \]
\end{sol}

\begin{exercise}{1.2.3}
    Use the generators and relations above to show that every element of $D_{2n}$ which is not a power of $r$ has order $2$.
	Deduce that $D_{2n}$ is generated by the two elements $s$ and $sr$, both of which have order $2$.
\end{exercise}
 
\begin{sol}
    Since every element that is not a power of $r$ is of the form $s r^k$, it follows from \exref{1.2.1} that such elements have order $2$.
    Therefore, $\abs{s} = \abs{sr} = 2$.
	Moreover, $r = s(sr)$, so $r, s \in \gen{s, s r}$, and therefore $\gen{s, sr} = D_{2n}$.
\end{sol}

\begin{exercise}{1.2.4}
    If $n = 2k$ is even and $n \geq 4$, show that $z = r^k$ is an element of order $2$ which commutes with all elements of $D_{2n}$.
	Show also that $z$ is the only nonidentity element of $D_{2n}$ which commutes with all elements of $D_{2n}$.
    \hint{cf. \exref{1.1.33}.}
\end{exercise}
   
\begin{sol}
    Since the elements $1, r, \ldots, r^{n-1}$ are all distinct and $0 < k < n$, we have $r^k \neq 1$.
	Moreover, $(r^k)^2 = r^n = 1$, hence $\abs{r^k} = 2$.
    In particular, $r^k = (r^k)\inv = r^{-k}$.
	Since $r^k$ commutes with other powers of $r$, consider $sr^j \in D_{2n}$ for $0 \leq j < n$.
	By applying \exref{1.2.2} $k$ times with $x = s r^j$, we have $r^k (s r^j) = (s r^j) r^{-k} = (s r^j) r^k$.

    Now let $w \in D_{2n}$ such that $wx = xw$ for all $x \in D_{2n}$.
	If $w = r^m$ for $0 < m < n$, then commuting with $s$ gives $w = w\inv$, which implies $m = k$ by \exref{1.1.33}.
	If $w = sr^m$ instead, then $r w = w r$.
    By \exref{1.2.2} with $x = w$, we have $r w = w r\inv$, so $w r = w r\inv$, and therefore $r^2 = 1$, contradicting $\abs{r} \geq 4$.
\end{sol}

\begin{exercise}{1.2.5}
    If $n$ is odd and $n \geq 3$, show that the identity is the only element of $D_{2n}$ which commutes with all elements of $D_{2n}$.
    \hint{cf. \exref{1.1.33}.}
\end{exercise}

\begin{sol}
    Let $x \in D_{2n}$ commute with all elements of $D_{2n}$.
    If $x = r^m$ for some $0 < m < n$, commuting with $s$ gives $r^m = r^{-m}$, which cannot occur by \exref{1.1.33}.
    If $x = s r^m$ for some $0 \leq m < n$, commuting with $r$ gives $r s r^m = s r^m r$, and \exref{1.2.2} gives $r s r^m = s r^m r\inv$.
    Then $r^2 = 1$, contradicting $\abs{r} \geq 3$, so $x = 1$.
\end{sol}

\begin{exercise}{1.2.6}
    Let $x$ and $y$ be elements of order $2$ in any group $G$.
	Prove that if $t = xy$ then $tx = xt\inv$ (so that if $n = \abs{xy} < \infty$ then $x, t$ satisfy the same relations in $G$ as $s, r$ do in $D_{2n}$).
\end{exercise}

\begin{sol}
    Since $\abs{x} = \abs{y} = 2$, we have $x = x\inv$ and $y = y\inv$.
	Then
    \[
    	tx = (xy)x = x(yx) = x(xy)\inv = x t\inv. \qedhere
    \]
\end{sol}

\begin{exercise}{1.2.7}
    Show that $\gen{a, b \mid a^2 = b^2 = (ab)^n = 1}$ gives a presentation for $D_{2n}$ in terms of the two generators $a = s$ and $b = sr$ of order $2$ computed in \exref{1.2.3} above.
    \hint{Show that the relations for $r$ and $s$ follow from the relations for $a$ and $b$ and, conversely, the relations for $a$ and $b$ follow from those for $r$ and $s$.}
\end{exercise}

\begin{sol}
    If $a^2 = b^2 = (ab)^n = 1$, put $r = ab$ and $s = a$.
    Then
    \[
    	r^n = (ab)^n = 1, \quad s^2 = a^2 = 1, \quad rs = (ab)a = a(ba) = a(b\inv a\inv) = a(ab)\inv = sr\inv.
    \]
    Conversely, if $r^n = s^2 = 1$ and $rs = sr\inv$, put $a = s$ and $b = sr$.
    Then
    \[
    	a^2 = s^2 = 1, \quad b^2 = (sr)^2 = s(rs)r = s(sr\inv)r = 1, \quad (ab)^n = (s(sr))^n = r^n = 1.
    \]
    The first direction shows $sr = a(ab) = b$ and the second direction shows $ab = s(sr) = r$.
    Therefore, with $a = s$ and $b = sr$ as in \exref{1.2.3}, the presentations $\gen{a, b \mid a^2 = b^2 = (ab)^n = 1}$ and $\gen{r, s \mid r^n = s^2 = 1, rs = sr\inv}$ are equivalent, and thus both present the group $D_{2n}$.
\end{sol}

\begin{exercise}{1.2.8}
    Find the order of the cyclic subgroup of $D_{2n}$ generated by $r$ (cf. \exref{1.1.27}).
\end{exercise}

\begin{sol}
    Distinctness of the elements $1, r, r^2, \dots, r^{n-1}$ and the fact that $r^n = 1$ show that the cyclic subgroup generated by $r$ has order $n$.
\end{sol}

In each of Exercises 9 to 13 you can find the order of the group of rigid motions in $\rbb^3$ (also called the group of rotations) of the given Platonic solid by following the proof for the order of $D_{2n}$:
find the number of positions to which an adjacent pair of vertices can be sent.
Alternatively, you can find the number of places to which a given face may be sent and, once a face is fixed, the number of positions to which a vertex on that face may be sent.

\begin{exercise}{1.2.9} 
    Let $G$ be the group of rigid motions in $\rbb^3$ of a tetrahedron.
	Show that $\abs{G} = 12$.
\end{exercise}

\begin{sol}
    Label the vertices of the tetrahedron $1$ through $4$.
    Vertex $1$ has $4$ choices, and vertex $2$ has $3$ choices.
    Since symmetries are rigid motions, the positions of the remaining vertices are determined once vertices $1$ and $2$ are fixed, so $\abs{G} = 4 \cdot 3 = 12$.
\end{sol}

\begin{exercise}{1.2.14}
    Find a set of generators for $\zbb$.
\end{exercise}

\begin{sol}
    Any integer is of the form $n1$ with $n \in \zbb$, so $\zbb = \gen 1$.
\end{sol}

\begin{exercise}{1.2.15}
    Find a set of generators and relations for $\intmod$.
\end{exercise}

\begin{sol}
    Since $n \bar 1 = \bar 0$, every element in $\intmod$ is of the form $k \bar 1$ for some $0 \leq k < n$, and these are distinct by \exref{0.3.2}, so $\intmod = \gen{\bar 1 \mid n \bar 1 = \bar 0}$.
\end{sol}

\begin{exercise}{1.2.16}
    Show that the group $\gen{x_1, y_1 \mid x_1^2 = y_1^2 = (x_1 y_1)^2 = 1}$ is the dihedral group $D_4$ (where $x_1$ may be replaced by the letter $r$ and $y_1$ by $s$).
\hint{Show that the last relation is the same as: $x_1 y_1 = y_1 x_1\inv$.}
\end{exercise}

\begin{sol}
    Since $y_1^2 = 1$, we have $y_1\inv = y_1$, so
    \[
        (x_1 y_1)^2 = 1 \iff x_1 y_1 x_1 y_1 = 1 \iff x_1 y_1 x_1 = y_1\inv = y_1 \iff x_1 y_1 = y_1 x_1\inv.
    \]
    Therefore, the relations above are equivalent to the relations of $D_4$ with $r = x_1$ and $s = y_1$.
\end{sol}

\begin{exercise}{1.2.17}
    Let $X_{2n}$ be the group whose presentation is displayed in (1.2).
    \begin{subexercises}
        \item Show that if $n = 3k$, then $X_{2n}$ has order $6$, and it has the same generators and relations as $D_6$ when $x$ is replaced by $r$ and $y$ by $s$.
        \item Show that if $(3, n) = 1$, then $x$ satisfies the additional relation: $x = 1$.
	    In this case deduce that $X_{2n}$ has order $2$.
        \hint{Use the facts that $x^n = 1$ and $x^3 = 1$.}
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item As shown in the text, $x^3 = 1$.
	Assume the relations of $X_{6k}$ hold, and put $r = x$ and $s = y$.
    Then
    \[
        r^3 = x^3 = 1, \quad s^2 = y^2 = 1, \quad rs = xy = yx^2 = sr^2 = sr\inv.
    \]
    Now suppose the relations of $D_6$ hold, and put $x = r$ and $y = s$, so
    \[
        x^{3k} = (r^3)^k = 1, \quad y^2 = s^2 = 1, \quad xy = rs = sr^2 = yx^2.
    \]
    Therefore, the two presentations are equivalent, and $\abs{X_{6k}} = \abs{D_6} = 6$.
    \item Since $(3, n) = 1$, there exist integers $a$ and $b$ such that $3a + nb = 1$.
	From the relations of $X_{2n}$ and as in (a), $x^n = 1$ and $x^3 = 1$, so
    \[
    	x = x^{3a + nb} = (x^3)^a (x^n)^b = 1^a 1^b = 1.
    \]
    Since $x = 1$, the only remaining relation is $y^2 = 1$, and thus $X_{2n} = \set{1, y}$ with $\abs{X_{2n}} = 2$.
\end{solenum}

\begin{exercise}{1.2.18}
    Let $Y$ be the group whose presentation is displayed in (1.3).
    \begin{subexercises}
        \item Show that $v^2 = v\inv$.
        \hint{Use the relation: $v^3 = 1$.}
        \item Show that $v$ commutes with $u^3$.
        \hint{Show that $v^2 u^3v = u^3$ by writing the left hand side as $(v^2u^2)(uv)$ and using the relations to reduce this to the right hand side.
	    Then use part (a).}
        \item Show that $v$ commutes with $u$.
        \hint{Show that $u^9 = u$ and then use part (b).}
        \item Show that $uv = 1$.
        \hint{Use part (c) and the last relation.}
        \item Show that $u = 1$, deduce that $v = 1$, and conclude that $Y = 1$.
        \hint{Use part (d) and the equation $u^4 v^3 = 1$.}
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item $v^3 = 1 \implies v^2 = v\inv$.
    \item From $uv = v^2u^2$, we have $vuv = u^2$ so that $vu = u^2v\inv = u^2v^2$.
    It follows that
    \[
        v^2u^3v = (v^2u^2)(uv) = (uv)(uv) = u(vu)v = u(u^2v^2)v = u^3.
    \]
    By part (a), $v\inv = v^2$, so $v\inv u^3 v = u^3$, and thus $v$ commutes with $u^3$.
    \item Since $u^4 = 1$, we have $u^9 = (u^4)^2u = u$, so part (b) gives $uv = u^9v = (u^3)^3v = v(u^3)^3 = vu$.
    \item We have $uv = v^2u^2 = u^2v^2$, and cancellation gives $1 = uv$.
    \item By part (c), $u^4v^3 = u(uv)^3$, and $u(uv)^3 = u1^3 = u$ by part (d). 
    The relations $u^4 = v^3 = 1$ then give $u = 1$, and part (d) gives $v = 1$, so $Y = 1$.
\end{solenum}
