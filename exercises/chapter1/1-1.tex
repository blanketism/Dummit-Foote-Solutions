\section{Introduction to Groups}

\subsection{Basic Axioms and Examples}

Let $G$ be a group.

\begin{exercise}{1.1.1}
    Determine which of the following binary operations are associative.
    \begin{subexercises}
        \item The operation $\star$ on $\zbb$ defined by $a \star b = a - b$.
        \item The operation $\star$ on $\rbb$ defined by $a \star b = a + b + ab$.
        \item The operation $\star$ on $\qbb$ defined by
        \[
        	a \star b = \dfrac{a + b}{5}.
        \]
        \item The operation $\star$ on $\zbb \times \zbb$ defined by $(a,b) \star (c,d) = (ad + bc, \, bd)$.
        \item The operation $\star$ on $\qbb - \set{0}$ defined by
        \[
        	a \star b = \dfrac{a}{b}.
        \]
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Not associative: $(1 \star 0) \star 1 = 0 \neq 2 = 1 \star (0 \star 1)$.
    \item Associative.
    \item Not associative: $(1 \star 0) \star 2 = 11/25 \neq 7/25 = 1 \star (0 \star 2)$.
    \item Associative.
    \item Not associative: $(1 \star 2) \star 3 = 1/6 \neq 3/2 = 1 \star (2 \star 3)$.
\end{solenum}

\begin{exercise}{1.1.3}
    Prove that addition of residue classes in $\intmod$ is associative (you may assume it is well-defined).
\end{exercise}

\begin{sol}
    Let $\bar a, \bar b, \bar c \in \intmod$.
	By associativity of addition in $\zbb$,
    \[
    	(\bar a + \bar b) + \bar c = \bar{(a + b) + c} = \bar{a + (b + c)} = \bar a + (\bar b + \bar c). \qedhere
    \]
\end{sol}

\begin{exercise}{1.1.4}
    Prove that multiplication of residue classes in $\intmod$ is associative (you may assume it is well-defined).
\end{exercise}

\begin{sol}
    Use the same argument as in the previous exercise, replacing $+$ with $\cdot$.
\end{sol}

\begin{exercise}{1.1.5}
    Prove for all $n > 1$ that $\intmod$ is not a group under multiplication of residue classes.
\end{exercise}

\begin{sol}
    Since $0 \cdot a = 0$ for all $a \in \zbb$, $\bar a \cdot \bar 0 = \bar 0$.
	Since $n > 1$, $\bar 0 \neq \bar 1$, so $\bar 0$ does not have a multiplicative inverse in $\intmod$.
	Hence, $\intmod$ is not a group under multiplication of residue classes.
\end{sol}

\begin{exercise}{1.1.6}
    Determine which of the following sets are groups under addition:
    \begin{subexercises}
        \item the set of rational numbers (including $0 = 0/1$) in lowest terms whose denominators are odd
        \item the set of rational numbers (including $0 = 0/1$) in lowest terms whose denominators are even
        \item the set of rational numbers of absolute value $< 1$
        \item the set of rational numbers of absolute value $\ge 1$ together with $0$
        \item the set of rational numbers with denominators equal to $1$ or $2$
        \item the set of rational numbers with denominators equal to $1$, $2$, or $3$
    \end{subexercises}
\end{exercise}

\begin{sol}
    Let $S$ denote the set in each part.
    \begin{subexercises}
        \item $S$ has identity element $0$, inverse element $-a/b$ for $a/b \in S$, and inherits associativity from $\qbb$.
	    Moreover, let $a/b, c/d \in S$. Then $b$ and $d$ are odd, so
        \[
        	\frac ab + \frac cd = \frac{ad + bc}{bd}
        \]
        with $bd$ odd.
	    Since the reduced denominator divides $bd$, it must also be odd.
	    Hence, $S$ is closed under addition, and it is a group.
        \item No: $1/2 \in S$, but $1/2 + 1/2 = 2/2 = 1/1 \notin S$.
        \item No: $1/2 \in S$, but $1/2 + 1/2 = 1 \notin S$.
        \item No: $3/2, -1 \in S$, but $3/2 + (-1) = 1/2 \notin S$.
        \item Note that $S = \set{x/2 \mid x \in \zbb}$.
        Then for $\frac{a}{2}, \frac{b}{2} \in S$, we have $\frac{a}{2} + \frac{b}{2} = \frac{a + b}{2} \in S$, so $S$ is closed under addition.
        Identity, inverses, and associativity are satisfied, so $(S, +)$ is a group.
        \item No: $1/3, 1/2 \in S$, but $1/3 + 1/2 = 5/6 \notin S$. \qedhere
    \end{subexercises}
\end{sol}

\begin{exercise}{1.1.7}
    Let $G = \set{ x \in \rbb \mid 0 \le x < 1 }$ and for $x, y \in G$ let $x \star y$ be the fractional part of $x + y$ (i.e., $x \star y = x + y - [x + y]$ where $[a]$ is the greatest integer less than or equal to $a$).
	Prove that $\star$ is a well-defined binary operation on $G$ and that $G$ is an abelian group under $\star$ (called the \term{real numbers mod $1$}).
\end{exercise}

\begin{sol}
    Let $x, y, z \in G$.
    \begin{itemize}
        \item Note that for any $a \in \rbb$, we have $[a] \leq a < [a] + 1$, so $0 \leq a - [a] < 1$.
        Applying this to $x + y$, we have $0 \leq x + y - [x + y] < 1$, so $x \star y \in G$, and $\star$ is well-defined.
        \item Observe that $x \star y = x + y - [x + y] = y + x - [y + x] = y \star x$, so $G$ is abelian under $\star$.
        \item Because $x \star 0 = x + 0 - [x + 0] = x - [x] = x$, it follows that $0$ is the identity element of $G$ under $\star$.
        \item Since $0 \star 0 = 0$, the inverse of $0$ under $\star$ is itself.
	    For $x \neq 0$, note that $0 < x < 1$ gives $0 < 1 - x < 1$, so $1 - x \in G$.
        Therefore, the inverse of $x$ is $1 - x$, since $x \star (1 - x) = x + (1 - x) - [x + (1 - x)] = 0$.
        \item Note that $[\alpha + \beta] = [\alpha] + \beta$ for $\beta \in \zbb$.
	    Then
        \begin{align*}
            (x \star y) \star z & = (x + y - [x + y]) \star z \\
            & = x + y + z - [x + y] - [x + y + z - [x + y]] \\
            & = x + y + z - [x + y] - ([x + y + z] - [x + y]) \\
            & = x + y + z - [x + y + z] \\
            & = x \star (y + z - [y + z]) = x \star (y \star z). \qedhere
        \end{align*}
    \end{itemize}
\end{sol}

\begin{exercise}{1.1.8}
    Let $G = \set{z \in \cbb\mid z^n = 1 \text{ for some } n \in \zp }$.
    \begin{subexercises}
        \item Prove that $G$ is a group under multiplication (called the group of \term{roots of unity} in $\cbb$).
        \item Prove that $G$ is not a group under addition.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Since $1^1 = 1$, $1 \in G$.
	If $z \in G$, then $z^n = 1$ for some $n \in \zp$, so $z \neq 0$ and $z\inv$ exists.
    Therefore, $(z\inv)^n = (z^n)\inv = 1\inv = 1$, so $z\inv \in G$.
	Associativity from $\cbb$ is inherited by $G$.
	Finally, let $z, w \in G$ such that $z^n = w^m = 1$ for $n, m \in \zp$.
	Then
    \[
    	(zw)^{nm} = z^{nm} w^{nm} = (z^n)^m (w^m)^n = 1 \cdot 1 = 1,
    \]
    so that $zw \in G$.
	Hence, $G$ is a group under multiplication.
    \item No; note $1, -1 \in G$, but $1 + (-1) = 0 \notin G$ since $0^n = 0 \neq 1$ for any $n \in \zp$.
\end{solenum}

\begin{exercise}{1.1.9}
    Let $G = \set{ a + b\sqrt{2} \in \rbb \mid a, b \in \qbb }$.
    \begin{subexercises}
        \item Prove that $G$ is a group under addition.
        \item Prove that the nonzero elements of $G$ are a group under multiplication.
        
        \hint{``Rationalize the denominators'' to find multiplicative inverses.}
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item For any $a + b\sqrt{2}, c + d\sqrt{2} \in G$, we have $ (a + b\sqrt{2}) + (c + d\sqrt{2}) = (a + c) + (b + d)\sqrt{2} \in G$, so $G$ is closed under addition.
	The identity is $0 + 0\sqrt 2 = 0$, and the inverse of $a + b\sqrt 2$ is $-a - b\sqrt 2$.
	Associativity is inherited from $\rbb$.
	Hence, $G$ is a group under addition.
    \item Suppose $a + b\sqrt{2}, c + d\sqrt{2} \in G$ are nonzero.
    Then $(a + b\sqrt 2)(c + d\sqrt 2) = (ac + 2bd) + (ad + bc)\sqrt{2}$.
    Since a product of two nonzero reals is nonzero, $(a + b\sqrt 2)(c + d\sqrt 2)$ is nonzero.
    Moreover, $ac + 2bd, ad + bc \in \qbb$, so $(a + b\sqrt 2)(c + d\sqrt 2) \in G$.
	The identity is $1 + 0\sqrt 2 = 1$, and the inverse of a nonzero element $a + b\sqrt 2$ is
    \[
    	(a + b\sqrt 2)\inv = \frac{1}{a + b\sqrt 2} = \frac{1}{a + b\sqrt 2} \cdot \frac{a - b\sqrt 2}{a - b\sqrt 2} = \frac{a - b\sqrt 2}{a^2 - 2b^2} = \frac a{a^2 - 2b^2} - \frac b{a^2 - 2b^2}\sqrt 2.
    \]
    If $a - b\sqrt{2} = 0$ and $b = 0$, then $a = 0$, contradicting that $a + b\sqrt{2}$ is nonzero.
    If $a - b\sqrt{2} = 0$ and $b \neq 0$, then $a = b\sqrt{2}$, contradicting that $a \in \qbb$.
    Therefore, $a - b\sqrt{2} \neq 0$, so $a^2 - 2b^2 = (a - b\sqrt{2})(a + b\sqrt{2}) \neq 0$ since it is a product of two nonzero reals.
    Hence, the denominator is nonzero and the inverse exists in $G$.
    Since associativity is inherited from $\rbb$, the nonzero elements of $G$ form a group under multiplication.
\end{solenum}

\begin{exercise}{1.1.10}
    Prove that a finite group is abelian if and only if its group table is a symmetric matrix.
\end{exercise}

\begin{sol}
    Let $G = \set{g_1, g_2, \ldots, g_n}$ be a finite group.
    The $(i,j)$th and $(j,i)$th entries of the group table are $g_i g_j$ and $g_j g_i$, respectively.
    Then
    \[
        \text{the group table is symmetric} \iff g_i g_j = g_j g_i \text{ for all } i, j \iff G \text{ is abelian}. \qedhere
    \]
\end{sol}

\begin{exercise}{1.1.15}
    Prove that $(a_1 a_2 \ldots a_n)^{-1} = a_n^{-1} a_{n-1}^{-1} \ldots a_1^{-1}$ for all $a_1, a_2, \ldots, a_n \in G$.
\end{exercise}

\begin{sol}
    We proceed by induction on $n$.
    The base case $n = 1$ holds trivially since $(a_1)\inv = a_1\inv$.
	Suppose it holds for some $n \in \zp$.
	By Proposition 1.1(4), (5), and the inductive hypothesis, we have
    \[
    	(a_1 a_2 \cdots a_n a_{n+1})^{-1} = a_{n+1}^{-1} (a_1 a_2 \cdots a_n)^{-1} = a_{n+1}^{-1} a_n^{-1} a_{n-1}^{-1} \cdots a_1^{-1}. \qedhere
    \]
\end{sol}

\begin{exercise}{1.1.16}
    Let $x$ be an element of $G$.
	Prove that $x^2 = 1$ if and only if $|x|$ is either $1$ or $2$.
\end{exercise}

\begin{soliff}
    \rightimp Suppose $x^2 = 1$.
	Then $|x| \leq 2$.
	Since $|x| \in \zp$, $|x|$ is either $1$ or $2$.

    \leftimp If $|x| = 1$, then $x = 1$ and $x^2 = 1$.
	If $|x| = 2$, then $x^2 = 1$ by definition.
\end{soliff}

\begin{exercise}{1.1.17}
    Let $x$ be an element of $G$.
	Prove that if $|x| = n$ for some positive integer $n$, then $x^{-1} = x^{n-1}$.
\end{exercise}

\begin{sol}
    Note that $x^n = x^{n - 1}x = 1$, so $x^{n - 1} = x\inv$.
\end{sol}

\begin{exercise}{1.1.18}
    Let $x, y$ be elements of $G$.
	Prove that the following are equivalent:

    (i) $xy = yx$; \quad (ii) $y\inv xy = x$; \quad (iii) $x\inv y\inv xy = 1$.
\end{exercise}

\begin{sol}
    $xy = yx \iff y\inv xy = y\inv yx = x \iff x\inv y\inv xy = x\inv x = 1$.
\end{sol}

\begin{exercise}![Exponent Laws in Groups]{1.1.19}
    Let $x \in G$ and let $a, b \in \zp$.
    \begin{subexercises}
        \item Prove that $x^{a+b} = x^a x^b$ and $(x^a)^b = x^{ab}$.
        \item Prove that $(x^a)^{-1} = x^{-a}$.
        \item Establish part (a) for arbitrary integers $a$ and $b$.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item By Proposition 1.1(5), both $x^{a+b}$ and $x^a x^b$ have ${a+b}$ copies of $x$, up to bracketing, and thus $x^{a+b} = x^a x^b$.
    Similarly, both $(x^a)^b$ and $x^{ab}$ have $ab$ copies of $x$, up to bracketing, and thus $(x^a)^b = x^{ab}$.
    \item By \exref{1.1.15}, $(x^a)\inv$ consists of $a$ instances of $x\inv$, and therefore $(x^a)\inv = x^{-a}$.
    In particular, $(x^n)\inv = x^{-n}$ for all $n \in \zbb$, where $n > 0$ is the preceding argument.
    The case of $n = 0$ is trivial since $x^0 = 1$ and $1\inv = 1$, and if $n < 0$, then applying the preceding argument to $-n > 0$ and Proposition 1.1(3) gives $(x^n)\inv = x^{-n}$.
    \item Suppose one of $a$ or $b$ is $0$.
    Then $x^{a+b} = x^a x^b$ trivially holds since $x^0 = 1$.
    Moreover, $(x^0)^b = 1^b = 1 = x^{0 \cdot b}$ for any integer $b$, and $(x^a)^0 = 1 = x^{a \cdot 0}$ for any integer $a$.
    Otherwise, consider the cases:
    \begin{itemize}
        \item If $a, b \in \zp$, then the result follows from part (a).
        \item If $a \in \zp$ and $b < 0$, suppose $a + b \geq 0$.
        Then
        \[
            x^a = x^{a+b-b} = x^{a+b} x^{-b} = x^{a+b} (x^b)\inv,
        \]
        where the last equality follows from part (b).
        Thus, $x^{a+b} = x^a x^b$.
        If $a + b < 0$, applying the preceding result to $-a < 0$ and $-b \in \zp$ gives $x^{-(a+b)} = x^{-b} x^{-a}$.
        By part (b), taking inverses gives
        \[
            (x^{a+b})\inv = x^{-(a+b)} = x^{-b} x^{-a} = (x^a x^b)\inv,
        \]
        and uniqueness of inverses gives $x^{a+b} = x^a x^b$.
        The case of $a < 0$ and $b \in \zp$ is analogous to the previous case with roles of $a$ and $b$ swapped.

        For the multiplication law, if $a \in \zp$ and $b < 0$, then
        \[
            (x^a)^b = ((x^a)^{-b})\inv = (x^{a(-b)})\inv = (x^{-ab})\inv = x^{ab}.
        \]
        If $a < 0$ and $b \in \zp$, then
        \[
            (x^a)^b = ((x^{-a})\inv)^b = ((x^{-a})^b)\inv = x^{ab}.
        \]
        \item If $a < 0$ and $b < 0$, then $a + b < 0$ and
        \[
        	x^{a + b} = x^{b + a} = (x^{-b - a})\inv = (x^{-b} x^{-a})\inv = (x^{-a})\inv (x^{-b})\inv = x^a x^b.
        \]
        Moreover, $((x^{-a})\inv)^b = (x^{-a})^{-b} = ((x^{-a})^b)\inv$ follows from the definition of a negative exponent, Proposition 1.1(3), and part (b).
        Therefore,
        \[
            (x^a)^b = ((x^{-a})\inv)^b = ((x^{-a})^b)\inv = (x^{(-a)b})\inv = (x^{-ab})\inv = x^{ab}. \qedhere
        \]
    \end{itemize}
\end{solenum}

\begin{exercise}{1.1.20}
    For $x$ an element in $G$, show that $x$ and $x^{-1}$ have the same order.
\end{exercise}

\begin{sol}
    Suppose $|x| = n \in \zp$.
	Then
    \[
        (x\inv)^n = x^{-n} = (x^n)\inv = 1\inv = 1,
    \]
    so $|x\inv| \leq n$.
    Since $(x\inv)\inv = x$, applying the preceding argument to $x\inv$ gives $n \leq |x\inv|$, and therefore $|x\inv| = n = |x|$.
	In particular, $x$ has finite order if and only if $x\inv$ has finite order, and hence $x$ has infinite order if and only if $x\inv$ has infinite order.
\end{sol}

\begin{exercise}{1.1.21}
    Let $G$ be a finite group and let $x$ be an element of order $n$.
	Prove that if $n$ is odd, then $x = (x^2)^k$ for some $k \geq 1$.
\end{exercise}

\begin{sol}
    Put $n = 2k - 1$ for $k \in \zp$.
	Then $x^n = x^{2k - 1} = 1$, so that $x^{2k} = x$.
	Hence, $x = (x^2)^k$.
\end{sol}

\begin{exercise}!{1.1.22}
    If $x$ and $g$ are elements of the group $G$, prove that $|x| = |g^{-1} x g|$.
	Deduce that $|ab| = |ba|$ for all $a, b \in G$.
\end{exercise}

\begin{sol}
    A simple induction on $n$ gives $(g\inv x g)^n = g\inv x^n g$ for all $n \in \zp$.
	Let $|x| = m \in \zp$.
	Then
    \[
    	(g\inv x g)^m = g\inv x^m g = g\inv 1 g = 1,
    \]
    so that $|g\inv x g| \leq m$.
	Note that $x = (g\inv)\inv(g\inv x g) g\inv$, so applying the preceding argument with $g\inv$ in place of $g$ and $g\inv x g$ in place of $x$ gives $m \leq |g\inv x g|$.
    Therefore, $|g\inv x g| = m = |x|$, so $x$ has finite order if and only if $g\inv x g$ has finite order, and $x$ has infinite order if and only if $g\inv x g$ has infinite order.

    In particular, for any $a, b \in G$, putting $x = ab$ and $g = a$ gives $|ab| = |a\inv (ab) a| = |ba|$.
\end{sol}

\begin{exercise}{1.1.23}
    Suppose $x \in G$ and $|x| = n < \infty$.
	If $n = st$ for some positive integers $s$ and $t$, prove that $|x^s| = t$.
\end{exercise}

\begin{sol}
    Since $(x^s)^t = x^{st} = x^n = 1$, we have $|x^s| \leq t$.
	Moreover, if $(x^s)^r = 1$ for some $r \in \zp$, then $x^{sr} = 1$, so $n = st \leq sr$, hence $t \leq r$.
	Therefore, $|x^s| = t$.
\end{sol}

\begin{exercise}{1.1.24}
    If $a$ and $b$ are commuting elements of $G$, prove that $(ab)^n = a^n b^n$ for all $n \in \zbb$. 
    
    \hint{Do this by induction for positive $n$ first.}
\end{exercise}

\begin{sol}
    We prove the statement by induction on $n \in \zp$.
    The case $n = 1$ is clear since $(ab)^1 = ab = a^1 b^1$.
    Suppose it holds for some $n \in \zp$.
    Noting that $b^n a = a b^n$ by another short induction, we have
    \[
        (ab)^{n+1} = (ab)^n (ab) = (a^n b^n)(ab) = a^n (b^n a) b = a^n (ab^n) b = a^{n+1} b^{n+1}.
    \]
    If $n = 0$, then $(ab)^0 = 1 = a^0 b^0$.
    For $n < 0$, note that $-n > 0$.
    By the positive case,
    \[
        (ab)^n = ((ab)^{-n})\inv = (a^{-n} b^{-n})\inv = (b^{-n})\inv (a^{-n})\inv = b^n a^n = a^n b^n. \qedhere
    \]
\end{sol}

\begin{exercise}{1.1.25}
    Prove that if $x^2 = 1$ for all $x \in G$, then $G$ is abelian.
\end{exercise}

\begin{sol}
    Since $x^2 = 1$, $x = x\inv$ for all $x \in G$.
	Let $a, b \in G$.
	Then
    \[
    	ab = (ab)\inv = b\inv a\inv = ba. \qedhere
    \]
\end{sol}

\begin{exercise}![Subgroups]{1.1.26}
    Assume $H$ is a nonempty subset of $(G, \star)$ which is closed under the binary operation on $G$ and is closed under inverses, i.e., for all $h, k \in H$, $hk, \, h^{-1} \in H$.
	Prove that $H$ is a group under the operation $\star$ restricted to $H$ (such a subset $H$ is called a \term{subgroup} of $G$).
\end{exercise}

\begin{sol}
    Let $h \in H$.
    By assumption, $h\inv \in H$, so $1 = h h\inv \in H$, whence $H$ contains the identity.
    Moreover, $\star$ is a binary operation on $H$ since $H$ is closed under $\star$.
    Associativity is inherited from $G$.
    Finally, every element of $H$ has an inverse in $H$ by assumption.
    Therefore, $(H, \star|_H)$ is a group.
\end{sol}

\begin{exercise}![Cyclic Subgroups]{1.1.27}
    Prove that if $x$ is an element of the group $G$, then $\set{x^n \mid n \in \zbb}$ is a subgroup of $G$ (called the \term{cyclic subgroup} of $G$ generated by $x$).
\end{exercise}

\begin{sol}
    Let $H = \set{x^n \mid n \in \zbb}$.
    Then $1 = x^0 \in H$, so that $H$ is nonempty.
    Let $x^m, x^n \in H$.
    Then $x^m x^n = x^{m+n} \in H$ and $(x^m)\inv = x^{-m} \in H$.
    Hence, $H$ is closed under the binary operation on $G$ and inverses.
    By \exref{1.1.26}, $H$ is a subgroup of $G$.
\end{sol}

\begin{exercise}{1.1.28}
    Let $(A, \star)$ and $(B, \diamond)$ be groups and let $A \times B$ be their direct product (as defined in Example 6).
	Verify all the group axioms for $A \times B$:
    \begin{subexercises}
        \item Prove that the associative law holds, i.e., for all $(a_1, b_1), (a_2, b_2), (a_3, b_3) \in A \times B$:
        \[
        	( (a_1, b_1)(a_2, b_2) )(a_3, b_3) = (a_1, b_1)((a_2, b_2)(a_3, b_3))
        \]
        \item Prove that $(1, 1)$ is the identity of $A \times B$, and
        \item Prove that the inverse of $(a, b)$ is $(a^{-1}, b^{-1})$.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Let $(a_1, b_1), (a_2, b_2), (a_3, b_3) \in A \times B$.
	Then
    \begin{align*}
        [(a_1, b_1)(a_2, b_2)](a_3, b_3) &= (a_1 \star a_2, b_1 \diamond b_2)(a_3, b_3) \\
        &= ((a_1 \star a_2) \star a_3, (b_1 \diamond b_2) \diamond b_3) \\
        &= (a_1 \star (a_2 \star a_3), b_1 \diamond (b_2 \diamond b_3)) \\
        &= (a_1, b_1)(a_2 \star a_3, b_2 \diamond b_3) \\
        &= (a_1, b_1)[(a_2, b_2)(a_3, b_3)].
    \end{align*}
    \item For $a \in A, b \in B$, we have $(a, b)(1, 1) = (a \star 1, b \diamond 1) = (a, b) = (1 \star a, 1 \diamond b) = (1, 1)(a, b)$.
    \item $(a, b)(a\inv, b\inv) = (a \star a\inv, b \diamond b\inv) = (1, 1) = (a\inv \star a, b\inv \diamond b) = (a\inv, b\inv)(a, b)$.
\end{solenum}

\begin{exercise}{1.1.29}
    Prove that $A \times B$ is an abelian group if and only if both $A$ and $B$ are abelian.
\end{exercise}

\begin{sol}
    For all $a, a' \in A$ and $b, b' \in B$, we have
    \[
        (a, b)(a', b') = (a', b')(a, b) \iff (aa', bb') = (a'a, b'b) \iff aa' = a'a \text{ and } bb' = b'b.
    \]
    Thus, $A \times B$ is abelian if and only if $A$ and $B$ are abelian.
\end{sol}

\begin{exercise}{1.1.30}
    Prove that the elements $(a, 1)$ and $(1, b)$ of $A \times B$ commute and deduce that the order of $(a, b)$ is the least common multiple of $|a|$ and $|b|$.
\end{exercise}

\begin{sol}
    Let $a \in A$ and $b \in B$.
	Then
    \[
    	(a, 1)(1, b) = (a \cdot 1, 1 \cdot b) = (a, b) = (1 \cdot a, b \cdot 1) = (1, b)(a, 1)
    \]
    so that $(a, 1)$ and $(1, b)$ commute.
	Note that $(a, b)^k = (a, 1)^k (1, b)^k = (a^k, b^k)$ for all $k \in \zp$ by \exref{1.1.24}.
    If $|(a, b)|$ is finite, then $(a^{|(a, b)|}, b^{|(a, b)|}) = (1, 1)$, whence $|a|$ and $|b|$ are both finite.
    Conversely, finite $|a|$ and $|b|$ imply that $|(a, b)|$ is finite since $(a, b)^{|a|\,|b|} = (a^{|a|\,|b|}, b^{|a|\,|b|}) = (1, 1)$.
    Therefore, $|(a, b)| = \infty$ if and only if $|a| = \infty$ or $|b| = \infty$.
    Now suppose $|(a, b)|$ is finite, and let $n = |(a, b)|$ and $m = \lcm(|a|, |b|)$.
    Then $(a, b)^m = (a^m, b^m) = (1, 1)$, so that $n \leq m$.
    Suppose $(a, b)^\ell = (1, 1)$ for some $\ell \in \zp$.
    By the division algorithm, write $\ell = q_1 |a| + r_1$ with $0 \leq r_1 < |a|$ and $\ell = q_2 |b| + r_2$ with $0 \leq r_2 < |b|$.
    Then
    \[
        (a, b)^\ell = (a^{q_1 |a| + r_1}, b^{q_2 |b| + r_2}) = (a^{r_1}, b^{r_2}).
    \]
    By minimality of $|a|$ and $|b|$, we must have $r_1 = r_2 = 0$, and thus $\ell$ is a multiple of both $|a|$ and $|b|$.
    Therefore, $m \mid \ell$, and since $\ell$ was arbitrary, setting $\ell = n$ gives $m \mid n$.
    Combining this with $n \leq m$, we get $n = m$.
\end{sol}

\begin{exercise}{1.1.31}
    Prove that any finite group $G$ of even order contains an element of order 2. 
    
    \hint{Let $t(G)$ be the set $\set{g \in G \mid g \neq g\inv}$.
	Show that $t(G)$ has an even number of elements and every non-identity element of $G - t(G)$ has order 2.}
\end{exercise}

\begin{sol}
    Let $g \in t(G)$.
    Since $(g\inv)\inv = g$, we have $g\inv \in t(G)$.
    We have $|\set{g, g\inv}| = 2$ since $g \neq g\inv$.
    If $g, h \in t(G)$ and $\set{g, g\inv} \cap \set{h, h\inv} \neq \emptyset$, then either $g = h$ or $g = h\inv$, in which case $\set{g, g\inv} = \set{h, h\inv}$, so the sets are either equal or disjoint.
    Hence, $t(G)$ can be partitioned into pairs $\set{g, g\inv}$, and thus has even cardinality.
    Since $G$ has even order, $G - t(G)$ has even cardinality.
    Since $1 = 1\inv$, $1 \notin t(G)$, so $|G - t(G)| \geq 2$, and therefore there exists some $k \in G - t(G)$ with $k \neq 1$.
    For such a $k$, we have $k = k\inv$, and thus $k^2 = 1$, so $|k| = 2$.
\end{sol}

\begin{exercise}{1.1.32}
    If $x$ is an element of finite order $n$ in $G$, prove that the elements $1, x, x^2, \dots, x^{n-1}$ are all distinct.
	Deduce that $|x| \leq |G|$.
\end{exercise}

\begin{sol}
	Suppose $x^a = x^b$ for some integers $0 \leq a < b < n$.
	Then
    \[
    	1 = x^b x^{-a} = x^{b - a},
    \]
    so $|x| \leq b - a < n$, which is a contradiction.
	Hence, the elements $1, x, x^2, \dots, x^{n-1}$ are all distinct.
	Therefore, $G$ contains $n$ distinct elements, so $|x| = n \leq |G|$.
\end{sol}

\begin{exercise}{1.1.33}
    Let $x$ be an element of finite order $n$ in $G$.
    \begin{subexercises}
        \item Prove that if $n$ is odd then $x^i \neq x^{-i}$ for all $i = 1, 2, \dots, n-1$.
        \item Prove that if $n = 2k$ and $1 \leq i < n$ then $x^i = x^{-i}$ if and only if $i = k$.
    \end{subexercises}
\end{exercise}

\begin{solenum}
    \item Since $x^n = 1$, we have $x^{-i} = x^{n-i}$.
    Additionally, $1 \leq n-i \leq n-1$, so odd $n$ gives $i \neq n-i$, whence $x^i \neq x^{n-i} = x^{-i}$ by \exref{1.1.32}.
    \item Since $n = 2k$, $i \neq k$ gives $i \neq n-i$, so $x^i \neq x^{-i}$ as in part (a).
	If $i = k$, then $x^{-k} = x^{n-k} = x^k = x^i$.
\end{solenum}

\begin{exercise}{1.1.34}
    If $x$ is an element of infinite order in $G$, prove that the elements $x^n, n \in \zbb$ are all distinct.
\end{exercise}

\begin{sol}
    If $x^a = x^b$ for some $a, b \in \zbb$ such that $a < b$, then $x^{b - a} = 1$, contradicting that $x$ has infinite order.
	Hence, the elements $x^n, n \in \zbb$ are all distinct.
\end{sol}

\begin{exercise}{1.1.35}
    If $x$ is an element of finite order $n$ in $G$, use the Division Algorithm to show that \emph{any} integral power of $x$ equals one of the elements in the set $\set{1, x, x^2, \dots, x^{n-1}}$ (so these are all the distinct elements of the cyclic subgroup (cf. \exref{1.1.27} above) of $G$ generated by $x$).
\end{exercise}

\begin{sol}
    Let $m \in \zbb$.
    By the Division Algorithm, there exist integers $q$ and $r$ such that $m = qn + r$ with $0 \leq r < n$.
    Then
    \[
        x^m = (x^n)^q x^r = x^r. \qedhere
    \]
\end{sol}

\begin{exercise}!{1.1.36}
    Assume $G = \set{1, a, b, c}$ is a group of order 4 with identity 1.
	Assume also that $G$ has no elements of order 4 (so by \exref{1.1.32}, every element has order $\leq 3$).
	Use the cancellation laws to show that there is a unique group table for $G$.
	Deduce that $G$ is abelian.
\end{exercise}

\begin{sol}
    Since $G$ has even order, one of $a, b, c$ has order 2 by \exref{1.1.31}.
	Without loss of generality, let $a^2 = 1$.

    We first show that $ab = c$.
	Indeed, if $ab = 1$, then $b = a\inv = a$, contradicting that $b \neq a$.
	If $ab = a$, then $b = 1$, contradicting that $b \neq 1$.
	If $ab = b$, then $a = 1$, contradicting that $a \neq 1$.
	Hence, $ab = c$.
	By cancellation, we have $b = a\inv c = ac$.
	By symmetry, $ba = c$ and $b = ca$.

    To see that $|b| = |c| = 2$, we first note that $b^2 = (ca)(ac) = c(a^2)c = c^2$.
	If $b^2 = b$, then $b = 1$, contradicting that $b \neq 1$.
    If $b^2 = c$, then $b^2 = ab$, yielding $b = a$, contradicting that $b \neq a$.
    If $b^2 = a$, then $b^3 = ab = c \neq 1$, contradicting that $|b| \leq 3$.
    Hence, $b^2 = c^2 = 1$ so that $|b| = |c| = 2$.
    Then $bc = (ac)c = a(c^2) = a$, and similarly, $cb = c(ca) = (c^2)a = a$.
	Therefore, the group table is uniquely determined as follows:
    \[
        \begin{array}{c|cccc}
            \star & 1 & a & b & c \\
            \hline
            1 & 1 & a & b & c \\
            a & a & 1 & c & b \\
            b & b & c & 1 & a \\
            c & c & b & a & 1 \\
        \end{array},
    \]
    from which we see that $G$ is abelian by \exref{1.1.10}.
\end{sol}
